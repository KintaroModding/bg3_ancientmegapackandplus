// 1. trigger - level loaded, not yet spawned
IF
LevelGameplayStarted((STRING)_Level, _)
AND
DB_AMP_Lootbox_LevelConfig(_Level, _, _)
AND
NOT DB_AMP_Lootbox_Spawned(_Level)
THEN
DB_AMP_Lootbox_Spawned(_Level);
PROC_AMP_Lootbox_StartSpawn(_Level);

// 2. start spawn - calculate random count
PROC
PROC_AMP_Lootbox_StartSpawn((STRING)_Level)
AND
DB_AMP_Lootbox_LevelConfig(_Level, (INTEGER)_Min, (INTEGER)_Max)
AND
IntegerSubtract(_Max, _Min, (INTEGER)_Range)
AND
IntegerSum(_Range, 1, (INTEGER)_RangeInclusive)
AND
Random(_RangeInclusive, (INTEGER)_RandOffset)
AND
IntegerSum(_Min, _RandOffset, (INTEGER)_Count)
AND
_Count > 0
THEN
DB_AMP_Lootbox_SpawnRemaining(_Level, _Count);
PROC_AMP_Lootbox_SpawnNext(_Level);

// handle case when min=0 and random gives 0
PROC
PROC_AMP_Lootbox_StartSpawn((STRING)_Level)
AND
DB_AMP_Lootbox_LevelConfig(_Level, (INTEGER)_Min, (INTEGER)_Max)
AND
IntegerSubtract(_Max, _Min, (INTEGER)_Range)
AND
IntegerSum(_Range, 1, (INTEGER)_RangeInclusive)
AND
Random(_RangeInclusive, (INTEGER)_RandOffset)
AND
IntegerSum(_Min, _RandOffset, (INTEGER)_Count)
AND
_Count <= 0
THEN
DB_NOOP(1);

// 3. spawn next - pick random position, create chest
PROC
PROC_AMP_Lootbox_SpawnNext((STRING)_Level)
AND
DB_AMP_Lootbox_SpawnRemaining(_Level, (INTEGER)_Remaining)
AND
_Remaining > 0
AND
DB_AMP_Lootbox_PositionCount(_Level, (INTEGER)_Total)
AND
Random(_Total, (INTEGER)_RandPos)
AND
IntegerSum(_RandPos, 1, (INTEGER)_Index)
AND
DB_AMP_Lootbox_Position(_Level, _Index, (REAL)_X, (REAL)_Y, (REAL)_Z)
AND
NOT DB_AMP_Lootbox_UsedPosition(_Level, _Index)
AND
RealSum(_Y, 0.6, (REAL)_YOffset)
AND
DB_AMP_Lootbox_Template((ITEMROOT)_Template)
AND
CreateAt(_Template, _X, _YOffset, _Z, 0, 0, "", _Chest)
AND
IntegerSubtract(_Remaining, 1, (INTEGER)_NewRemaining)
THEN
DB_AMP_Lootbox_UsedPosition(_Level, _Index);
NOT DB_AMP_Lootbox_SpawnRemaining(_Level, _Remaining);
DB_AMP_Lootbox_SpawnRemaining(_Level, _NewRemaining);
PROC_AMP_Lootbox_CasualUnlock((ITEM)_Chest);
PROC_AMP_Lootbox_SpawnNext(_Level);

// fallback: position already used, try again
PROC
PROC_AMP_Lootbox_SpawnNext((STRING)_Level)
AND
DB_AMP_Lootbox_SpawnRemaining(_Level, (INTEGER)_Remaining)
AND
_Remaining > 0
AND
DB_AMP_Lootbox_PositionCount(_Level, (INTEGER)_Total)
AND
Random(_Total, (INTEGER)_RandPos)
AND
IntegerSum(_RandPos, 1, (INTEGER)_Index)
AND
DB_AMP_Lootbox_UsedPosition(_Level, _Index)
THEN
PROC_AMP_Lootbox_SpawnNext(_Level);

// 4. done - cleanup remaining counter
PROC
PROC_AMP_Lootbox_SpawnNext((STRING)_Level)
AND
DB_AMP_Lootbox_SpawnRemaining(_Level, 0)
THEN
NOT DB_AMP_Lootbox_SpawnRemaining(_Level, 0);
