INITSECTION

// t4 statuses that gate action surge, ultra only
DB_AMP_ActionSurge_T4Status("AMP_NPC_STANDARD_T4_V4");
DB_AMP_ActionSurge_T4Status("AMP_NPC_IMPOSSIBLE_T4_V4");
DB_AMP_ActionSurge_T4Status("AMP_BOSS_STANDARD_T4_V4");
DB_AMP_ActionSurge_T4Status("AMP_BOSS_IMPOSSIBLE_T4_V4");

KBSECTION

// ultra just turned on, guard already-tracked t4 enemies
PROC
PROC_AMP_Ultra_Activate()
AND
DB_AMP_DiffV3_Tracked((CHARACTER)_Creature, (STRING)_Status)
AND
DB_AMP_ActionSurge_T4Status(_Status)
AND
HasActiveStatus(_Creature, "AMP_ACTION_SURGE_INFO", 0)
THEN
ApplyStatus(_Creature, "AMP_ACTION_SURGE_INFO", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);

// ultra already on, guard a newly tracked t4 enemy
IF
DB_AMP_DiffV3_Tracked((CHARACTER)_Creature, (STRING)_Status)
AND
DB_AMP_ActionSurge_T4Status(_Status)
AND
DB_AMP_Ultra_Active(1)
AND
HasActiveStatus(_Creature, "AMP_ACTION_SURGE_INFO", 0)
THEN
ApplyStatus(_Creature, "AMP_ACTION_SURGE_INFO", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);

EXITSECTION
ENDEXITSECTION
