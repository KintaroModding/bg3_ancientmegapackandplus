// autoloot toggle
IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_AL_Toggle", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
THEN
PROC_AMP_AL_ShowToggle(_Caster);

PROC
PROC_AMP_AL_ShowToggle((CHARACTER)_Host)
AND
NOT DB_AMP_AutoLoot_Disabled(1)
THEN
OpenMessageBoxYesNo(_Host, "AMP_AL_Toggle_AskDisable");

PROC
PROC_AMP_AL_ShowToggle((CHARACTER)_Host)
AND
DB_AMP_AutoLoot_Disabled(1)
THEN
OpenMessageBoxYesNo(_Host, "AMP_AL_Toggle_AskEnable");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_Toggle_AskDisable", 1)
THEN
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_Toggle_AskDisable", 0)
THEN
NOT DB_AMP_AutoLoot_Disabled(0);
DB_AMP_AutoLoot_Disabled(1);
PROC_AMP_AutoLoot_RemoveAll();
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_Toggle_AskEnable", 1)
THEN
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_Toggle_AskEnable", 0)
THEN
NOT DB_AMP_AutoLoot_Disabled(1);
DB_AMP_AutoLoot_Disabled(0);
PROC_AMP_AutoLoot_AddAll();
PROC_AMP_PlayModeEffect(_Host);

PROC
PROC_AMP_AutoLoot_RemoveAll()
AND
DB_Players(_P)
THEN
RemovePassive(_P, "AMP_AutoLoot_Toggle");

PROC
PROC_AMP_AutoLoot_AddAll()
AND
DB_Players(_P)
THEN
AddPassive(_P, "AMP_AutoLoot_Toggle");

PROC
PROC_AMP_Filter_ShowSell_Containers((CHARACTER)_Host)
AND
NOT DB_AMP_AS_Disabled(1)
AND
NOT DB_AMP_AS_SellContainers(1)
THEN
OpenMessageBoxYesNo(_Host, "AMP_AS_FilterContainers_Sell_Off");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AS_FilterContainers_Sell_On", 1)
THEN
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AS_FilterContainers_Sell_On", 0)
THEN
NOT DB_AMP_AS_SellContainers(1);
DB_AMP_AS_SellContainers(0);
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AS_FilterContainers_Sell_Off", 1)
THEN
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AS_FilterContainers_Sell_Off", 0)
THEN
NOT DB_AMP_AS_SellContainers(0);
DB_AMP_AS_SellContainers(1);
PROC_AMP_PlayModeEffect(_Host);

IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_AL_CampSend", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
THEN
OpenMessageBoxYesNo(_Caster, "AMP_AL_CampSend_All");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_All", 1)
THEN
SysClear("DB_AMP_AL_CampSend", 1);
DB_AMP_AL_CampSend(1);

MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_All", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_AL_CampSend_Junk");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_Junk", 1)
THEN
SysClear("DB_AMP_AL_CampSend", 1);
DB_AMP_AL_CampSend(2);
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_Junk", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_AL_CampSend_Cheap");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_Cheap", 1)
THEN
SysClear("DB_AMP_AL_CampSend", 1);
DB_AMP_AL_CampSend(3);
PROC_AMP_PlayModeEffect(_Host);

THEN
OpenMessageBoxYesNo(_Host, "AMP_AL_CampSend_Off");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AL_CampSend_Off", 1)
THEN
SysClear("DB_AMP_AL_CampSend", 1);
DB_AMP_AL_CampSend(0);
PROC_AMP_PlayModeEffect(_Host);
