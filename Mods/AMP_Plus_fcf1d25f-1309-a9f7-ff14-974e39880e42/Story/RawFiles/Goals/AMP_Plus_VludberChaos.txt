THEN
PROC_AMP_Plus_VludberDisrobe_DrainQueue(_Char);

PROC
PROC_AMP_Plus_VludberDisrobe_DrainQueue((CHARACTER)_Char)
AND
DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z)
THEN
NOT DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z);
PROC_AMP_Plus_VludberDisrobe_DrainQueue(_Char);

// rarity read off gold value, same scale as DB_AMP_Arcanizer_RarityValueRange. legendary band stays on

IF
StatusApplied((CHARACTER)_Char, "AMP_VLUDBER_CHAOS_DISROBE", _, _)
AND
IsCharacter(_Char, 1)
AND
GetPosition(_Char, _X, _Y, _Z)
AND
DB_AMP_Plus_VludberDisrobe_Slot(_Slot)
AND
GetEquippedItem((CHARACTER)_Char, _Slot, _Item)
AND
ItemGetGoldValue(_Item, (INTEGER)_Value)
AND
DB_AMP_Arcanizer_RarityValueRange("Legendary", (INTEGER)_MinLegendary, _)
AND
_Value < _MinLegendary
THEN
DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z);
RealtimeObjectTimerLaunch(_Char, "AMP_Plus_VludberDisrobe", 500);

IF
ObjectTimerFinished((CHARACTER)_Char, "AMP_Plus_VludberDisrobe")
AND
DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z)
THEN
Unequip((CHARACTER)_Char, (ITEM)_Item);

IF

AND
DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z)
THEN
NOT DB_AMP_Plus_VludberDropping(_Char, _Item, _X, _Y, _Z);

AND
GetTemplate(_Vludber, _Template)
AND
DB_AMP_Vludber_Template((CHARACTERROOT)_Template)
AND
NOT DB_AMP_Plus_VludberVolley(_Vludber, _)
THEN

IF
ObjectTimerFinished((CHARACTER)_Vludber, "AMP_Plus_VludberVolley")
AND
DB_AMP_Plus_VludberVolley(_Vludber, (INTEGER)_Left)
AND
_Left > 0
AND
IsDead((CHARACTER)_Vludber, 0)
AND

THEN
NOT DB_AMP_Plus_VludberVolley(_Vludber, _Left);
DB_AMP_Plus_VludberVolley(_Vludber, _Next);
UseSpell(_Vludber, "Shout_AMP_Vludber_ChaosVolley", _Vludber, _Vludber, 1);
RealtimeObjectTimerLaunch(_Vludber, "AMP_Plus_VludberVolley", 2500);

IF
ObjectTimerFinished((CHARACTER)_Vludber, "AMP_Plus_VludberVolley")
AND
DB_AMP_Plus_VludberVolley(_Vludber, 0)
THEN
NOT DB_AMP_Plus_VludberVolley(_Vludber, 0);

IF
LeftCombat((CHARACTER)_Vludber, _)
AND
DB_AMP_Plus_VludberVolley(_Vludber, (INTEGER)_Left)
THEN
NOT DB_AMP_Plus_VludberVolley(_Vludber, _Left);
RealtimeObjectTimerCancel(_Vludber, "AMP_Plus_VludberVolley");

IF
Died((CHARACTER)_Vludber)
AND
DB_AMP_Plus_VludberVolley(_Vludber, (INTEGER)_Left)
THEN
NOT DB_AMP_Plus_VludberVolley(_Vludber, _Left);
RealtimeObjectTimerCancel(_Vludber, "AMP_Plus_VludberVolley");
