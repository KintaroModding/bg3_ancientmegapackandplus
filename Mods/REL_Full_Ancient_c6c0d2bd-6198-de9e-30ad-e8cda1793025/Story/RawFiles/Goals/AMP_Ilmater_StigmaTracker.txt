// record caster -> target -> stigma on any ilmater stigma apply
IF
StatusApplied((CHARACTER)_Target, _StigmaType, (CHARACTER)_Caster, _)
AND
DB_AMP_Ilmater_StigmaType(_StigmaType)
THEN
DB_AMP_Ilmater_StigmaLink(_Caster, _Target, _StigmaType);

// drop the link when the stigma expires or is removed
IF
StatusRemoved((CHARACTER)_Target, _StigmaType, _, _)
AND
DB_AMP_Ilmater_StigmaType(_StigmaType)
AND
DB_AMP_Ilmater_StigmaLink(_Caster, _Target, _StigmaType)
THEN
NOT DB_AMP_Ilmater_StigmaLink(_Caster, _Target, _StigmaType);

// single-target stigmas: applying to a new target removes the caster's previous one of the same type
IF
StatusApplied((CHARACTER)_NewTarget, _StigmaType, (CHARACTER)_Caster, _)
AND
DB_AMP_Ilmater_SingleTargetStigma(_StigmaType)
AND
DB_AMP_Ilmater_StigmaLink(_Caster, _OldTarget, _StigmaType)
AND
_OldTarget != _NewTarget
THEN
RemoveStatus(_OldTarget, _StigmaType, _Caster);

// clear links when the target dies
IF
Died((CHARACTER)_Target)
AND
DB_AMP_Ilmater_StigmaLink(_Caster, _Target, _StigmaType)
THEN
NOT DB_AMP_Ilmater_StigmaLink(_Caster, _Target, _StigmaType);

EXITSECTION
ENDEXITSECTION
