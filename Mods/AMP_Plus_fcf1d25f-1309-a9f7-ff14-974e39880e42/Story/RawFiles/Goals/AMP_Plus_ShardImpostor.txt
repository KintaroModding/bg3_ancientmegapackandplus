// outside the generic tables, so the total migration never runs for it. Run it here, ahead of the apply rules

PROC
PROC_AMP_Plus_ShardImpostor_Migrate((ITEM)_Item)
AND
NOT DB_AMP_Shard_TotalCount(_Item, _)
AND
DB_AMP_Shard_Count(_Item, _, _)
AND
ItemGetGoldValue(_Item, _Value)
AND
DB_AMP_Shard_PriceRange(_Rarity, _Min, _Max)
AND
_Value >= _Min
AND
_Value <= _Max
AND
DB_AMP_Shard_SlotLimit(_Rarity, _Limit)
THEN
DB_AMP_Shard_TotalCount(_Item, _Limit);

AND

AND
GetEquippedItem(_Char, _Slot, (ITEM)_Item)
THEN
PROC_AMP_Plus_ShardImpostor_Migrate(_Item);
