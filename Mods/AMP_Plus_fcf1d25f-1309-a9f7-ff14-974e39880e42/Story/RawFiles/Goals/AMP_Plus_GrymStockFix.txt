INITSECTION

KBSECTION

// spells are re-seated on every spawn and load, so the marker gates them

PROC
PROC_AMP_Plus_GrymStockStrip()
AND
DB_AMP_GameMode("STOCK")
THEN
ApplyStatus(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, "AMP_PLUS_GRYM_STOCK", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);
RemoveSpell((CHARACTER)S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, "Shout_AMP_Grym_SummonDrones", 0);
RemoveSpell((CHARACTER)S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, "Target_AMP_AdamantineGolem_Pull", 0);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

IF
SavegameLoaded()
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

IF
LevelGameplayStarted("WLD_Main_A", _)
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

IF
WentOnStage(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, 1)
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

IF
EnteredCombat(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, _)
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

// a resurrect drops the marker, the story raises him between phases
IF
Resurrected(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255)
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_Plus_GrymStockStrip();

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "NORMAL")
THEN
PROC_AMP_Plus_GrymStockRelease();

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "HARD")
THEN
PROC_AMP_Plus_GrymStockRelease();

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "CASUAL")
THEN
PROC_AMP_Plus_GrymStockRelease();

PROC
PROC_AMP_Plus_GrymStockRelease()
THEN
RemoveStatus(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255, "AMP_PLUS_GRYM_STOCK", NULL_00000000-0000-0000-0000-000000000000);

EXITSECTION
ENDEXITSECTION
