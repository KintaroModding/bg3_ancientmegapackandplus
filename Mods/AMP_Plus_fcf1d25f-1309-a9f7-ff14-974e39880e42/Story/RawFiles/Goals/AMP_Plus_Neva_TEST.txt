Version 1
SubGoalCombiner SGC_AND
INITSECTION
// debug helper. SE console: oe AMP_TestSpawnNeva / oe AMP_TestKillNeva
// registers DB_AMP_MirrorChase_Mob so her peekaboo/atrophy engine fires on a test spawn

KBSECTION

IF
TextEvent("AMP_TestSpawnNeva")
AND
GetHostCharacter(_Host)
AND
GetPosition(_Host, _X, _Y, _Z)
AND
RealSum(_Z, 3.0, _SZ)
AND
CreateAt((GUIDSTRING)AMP_ULTRA_Summon_11_d5f7dcba-bbc3-4395-9fd2-7d488055e05f, _X, _Y, _SZ, 0, 1, "", (GUIDSTRING)_Mob)
THEN
DB_AMP_Neva_TestSpawn((CHARACTER)_Mob);
DB_AMP_MirrorChase_Mob((CHARACTER)_Mob, AMP_ULTRA_Summon_11_d5f7dcba-bbc3-4395-9fd2-7d488055e05f);
ApplyStatus(_Mob, "AMP_MIRROR_SUPPORT_FLAG", -1.0, 1, _Mob);
ApplyStatus(_Mob, "AMP_MIRROR_DIVINE_SUPPORT", -1.0, 1, _Mob);
ApplyStatus(_Mob, "AMP_MIRROR_WRATH", -1.0, 1, _Mob);
ApplyStatus(_Mob, "AMP_ULTRA_NPC_OVERLAY", -1.0, 1, _Mob);
ApplyStatus(_Mob, "AMP_ULTRA_SPAWN_VFX", 6.0, 1, _Mob);
SetCharacterLootable(_Mob, 0);
SetRelationTemporaryHostile((CHARACTER)_Mob, _Host);
EnterCombat(_Mob, _Host);

IF
TextEvent("AMP_TestKillNeva")
THEN
PROC_AMP_Plus_NevaTestDrainSpawn();

PROC
PROC_AMP_Plus_NevaTestDrainSpawn()
AND
DB_AMP_Neva_TestSpawn((CHARACTER)_Mob)
THEN
NOT DB_AMP_Neva_TestSpawn(_Mob);
RequestDeleteTemporary(_Mob);
PROC_AMP_Plus_NevaTestDrainSpawn();

EXITSECTION

ENDEXITSECTION
