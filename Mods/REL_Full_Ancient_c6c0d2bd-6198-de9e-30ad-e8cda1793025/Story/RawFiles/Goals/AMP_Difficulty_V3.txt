// average party level (accumulator pattern)

// iterate players, sum levels + count
PROC
PROC_AMP_CalcAvgPartyLevel_Iterate()
AND
DB_Players((CHARACTER)_Player)
AND
GetLevel(_Player, (INTEGER)_Lvl)
AND
DB_AMP_PartyLvl_Acc((INTEGER)_Count, (INTEGER)_Sum)
AND
IntegerSum(_Count, 1, (INTEGER)_NewCount)
AND
IntegerSum(_Sum, _Lvl, (INTEGER)_NewSum)
THEN
NOT DB_AMP_PartyLvl_Acc(_Count, _Sum);
DB_AMP_PartyLvl_Acc(_NewCount, _NewSum);

// reset acc, iterate, store result
PROC
PROC_AMP_UpdateAvgPartyLevel()
AND
DB_AMP_PartyLvl_Acc((INTEGER)_Old1, (INTEGER)_Old2)
THEN
NOT DB_AMP_PartyLvl_Acc(_Old1, _Old2);

PROC
PROC_AMP_UpdateAvgPartyLevel()
THEN
DB_AMP_PartyLvl_Acc(0, 0);
PROC_AMP_CalcAvgPartyLevel_Iterate();

// clear old avg
PROC
PROC_AMP_UpdateAvgPartyLevel()
AND
DB_AMP_PartyLvl_Acc((INTEGER)_Count, (INTEGER)_Sum)
AND
_Count > 0
AND
DB_AMP_AvgPartyLevel((INTEGER)_OldAvg)
THEN
NOT DB_AMP_AvgPartyLevel(_OldAvg);

// set new avg
PROC
PROC_AMP_UpdateAvgPartyLevel()
AND
DB_AMP_PartyLvl_Acc((INTEGER)_Count, (INTEGER)_Sum)
AND
_Count > 0
AND
IntegerDivide(_Sum, _Count, (INTEGER)_AvgLvl)
THEN
DB_AMP_AvgPartyLevel(_AvgLvl);

// per-npc tier calc + apply. efflvl = (npc_lvl + avgparty*2) / 3

// entry: compute efflvl, hand off to the ally-aware dispatch
PROC
PROC_AMP_DiffV3_ApplyToCreature((CHARACTER)_Creature, (STRING)_Mode)
AND
GetLevel(_Creature, (INTEGER)_NpcLvl)
AND
DB_AMP_AvgPartyLevel((INTEGER)_PartyLvl)
AND
IntegerProduct(_PartyLvl, 2, (INTEGER)_Weighted)
AND
IntegerSum(_NpcLvl, _Weighted, (INTEGER)_Sum)
AND
IntegerDivide(_Sum, 3, (INTEGER)_EffLvl)
THEN
PROC_AMP_DiffV3_DispatchAllyAware(_Creature, _Mode, _EffLvl);

// ally-aware dispatch: hostiles get full tier, allies/neutrals drop one tier (min t0)
// by pushing efflvl to the top of the next-lower bracket
