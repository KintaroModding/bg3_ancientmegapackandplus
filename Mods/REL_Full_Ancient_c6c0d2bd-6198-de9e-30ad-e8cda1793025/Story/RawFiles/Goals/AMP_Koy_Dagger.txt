// koy dagger: dark urge detect + status management, per tier

// uncommon (base)

IF
StatusApplied(_Character, "AMP_KOY_DAGGER_TECHNICAL", _, _)
AND
DB_Players((CHARACTER)_Character)
AND
IsTagged((CHARACTER)_Character, (TAG)REALLY_DARK_URGE_cd611d7d-b67d-42b4-a75c-a0c6091ef8a2, 1)
THEN
ApplyStatus(_Character, "AMP_KOY_DAGGER_DURGE_ACTIVE", -1.0, 1, _Character);

IF
StatusRemoved(_Character, "AMP_KOY_DAGGER_TECHNICAL", _, _)
AND
DB_Players((CHARACTER)_Character)
AND
HasAppliedStatus(_Character, "SLAYER_PLAYER", 0)
AND
HasAppliedStatus(_Character, "SLAYER_PLAYER_10", 0)
THEN
RemoveStatus(_Character, "AMP_KOY_DAGGER_DURGE_ACTIVE", _Character);
