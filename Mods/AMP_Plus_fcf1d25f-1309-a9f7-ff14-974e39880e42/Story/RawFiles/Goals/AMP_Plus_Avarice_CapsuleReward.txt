PROC
PROC_AMP_Plus_AvaCap_DrainPool()
AND
DB_AMP_Plus_AvaCap_Pool((ITEMROOT)_A, (STRING)_B, (INTEGER)_C)
THEN
NOT DB_AMP_Plus_AvaCap_Pool(_A, _B, _C);
PROC_AMP_Plus_AvaCap_DrainPool();

IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
DB_Players(_Char)
AND
NOT DB_AMP_Plus_AvaCap_PartyCombat(_CombatID)
THEN
DB_AMP_Plus_AvaCap_PartyCombat(_CombatID);

IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
IsBoss(_Char, 1)
AND
NOT DB_AMP_Plus_AvaCap_BossCombat(_CombatID)
THEN
DB_AMP_Plus_AvaCap_BossCombat(_CombatID);

IF
CombatEnded((GUIDSTRING)_CombatID)
AND
DB_AMP_Plus_AvaCap_PartyCombat(_CombatID)
AND
DB_AMP_Plus_AvaCap_BossCombat(_CombatID)
AND
DB_AMP_GenerousLoot_Enabled(1)
AND
GetHostCharacter((CHARACTER)_Host)
AND
HasAppliedStatus(_Host, "AMP_PLUS_AVARICE_REWARD_CD", 0)
AND
Random(100, (INTEGER)_Roll)
AND
DB_AMP_Plus_AvaCap_BossPct((INTEGER)_Pct)
AND
_Roll < _Pct
AND
Random(2, (INTEGER)_Extra)
THEN
PROC_AMP_Plus_AvaCap_ApplyCDAll();
PROC_AMP_Plus_AvaCap_GrantOne(_Host);
PROC_AMP_Plus_AvaCap_GrantExtra(_Host, _Extra);

IF
CombatEnded((GUIDSTRING)_CombatID)
AND
DB_AMP_Plus_AvaCap_PartyCombat(_CombatID)
AND
NOT DB_AMP_Plus_AvaCap_BossCombat(_CombatID)
AND
DB_AMP_GenerousLoot_Enabled(1)
AND
GetHostCharacter((CHARACTER)_Host)
AND
HasAppliedStatus(_Host, "AMP_PLUS_AVARICE_REWARD_CD", 0)
AND
Random(100, (INTEGER)_Roll)
AND
DB_AMP_Plus_AvaCap_MobPct((INTEGER)_Pct)
AND
_Roll < _Pct
THEN
PROC_AMP_Plus_AvaCap_ApplyCDAll();
PROC_AMP_Plus_AvaCap_GrantOne(_Host);

IF
CombatEnded((GUIDSTRING)_CombatID)
AND
DB_AMP_Plus_AvaCap_PartyCombat(_CombatID)
THEN
NOT DB_AMP_Plus_AvaCap_PartyCombat(_CombatID);

NOT DB_AMP_Plus_AvaCap_BossCombat(_CombatID);

PROC
PROC_AMP_Plus_AvaCap_GrantOne((CHARACTER)_Host)
AND
QRY_GetRandom("DB_AMP_Plus_AvaCap_Pool", 3, "DB_AMP_Plus_AvaCap_Pick")
AND
DB_AMP_Plus_AvaCap_Pick((ITEMROOT)_Template, (STRING)_Type, (INTEGER)_Idx)
THEN
NOT DB_AMP_Plus_AvaCap_Pick(_Template, _Type, _Idx);
TemplateAddTo(_Template, _Host, 1, 1);

PROC
PROC_AMP_Plus_AvaCap_GrantExtra((CHARACTER)_Host, (INTEGER)_Extra)
AND
_Extra == 1
THEN
PROC_AMP_Plus_AvaCap_GrantOne(_Host);

// marker goes on every player, the host included, since the gate reads the host
PROC
PROC_AMP_Plus_AvaCap_ApplyCDAll()
AND
DB_Players((CHARACTER)_P)
THEN
ApplyStatus(_P, "AMP_PLUS_AVARICE_REWARD_CD", 1800.0, 1, NULL_00000000-0000-0000-0000-000000000000);

// cooldown is the status itself. Never re-arm it on load, statuses are not restored yet and every load would grant a fresh one

PROC
PROC_AMP_Plus_AvaCap_SetOdds((INTEGER)_Mob, (INTEGER)_Boss)
AND
DB_AMP_Plus_AvaCap_MobPct((INTEGER)_OldMob)
AND
DB_AMP_Plus_AvaCap_BossPct((INTEGER)_OldBoss)
THEN
NOT DB_AMP_Plus_AvaCap_MobPct(_OldMob);
NOT DB_AMP_Plus_AvaCap_BossPct(_OldBoss);
DB_AMP_Plus_AvaCap_MobPct(_Mob);
DB_AMP_Plus_AvaCap_BossPct(_Boss);

IF
LevelGameplayStarted(_, _)
AND
NOT DB_AMP_Plus_AvaCap_MobPct(_)
THEN
DB_AMP_Plus_AvaCap_MobPct(40);
DB_AMP_Plus_AvaCap_BossPct(60);

IF
DB_AMP_GameMode("CASUAL")
THEN
PROC_AMP_Plus_AvaCap_SetOdds(50, 75);

IF
DB_AMP_GameMode("NORMAL")
THEN
PROC_AMP_Plus_AvaCap_SetOdds(40, 60);

IF
DB_AMP_GameMode("HARD")
THEN
PROC_AMP_Plus_AvaCap_SetOdds(25, 50);
