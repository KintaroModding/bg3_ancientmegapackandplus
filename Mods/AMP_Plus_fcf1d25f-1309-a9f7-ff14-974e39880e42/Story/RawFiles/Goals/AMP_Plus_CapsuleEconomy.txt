PROC
PROC_AMP_Plus_CapEcon_Apply()
AND
DB_AMP_Plus_CapEcon_Charges((INTEGER)_Want)
AND
DB_AMP_VoidCapsule_ChargesMaxActive((INTEGER)_Cur)
AND
_Cur != _Want
THEN
NOT DB_AMP_VoidCapsule_ChargesMaxActive(_Cur);
DB_AMP_VoidCapsule_ChargesMaxActive(_Want);

PROC_AMP_Plus_CapEcon_Apply()
AND
DB_AMP_Plus_CapEcon_Window((INTEGER)_Want)

AND

THEN
