Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION

// vanilla DB_PlayerInCamp, not a flag of our own: one written through GetHostCharacter reads false for every other character

QRY
QRY_AMP_Plus_CampGuard()
AND
DB_PlayerInCamp((CHARACTER)_P)
AND
NOT DB_AMP_Plus_CampGuardOff(1)
THEN
DB_NOOP(1);

IF
TextEvent("AMP_CampGuardOff")
AND
NOT DB_AMP_Plus_CampGuardOff(1)
THEN
DB_AMP_Plus_CampGuardOff(1);

IF
TextEvent("AMP_CampGuardOn")
AND
DB_AMP_Plus_CampGuardOff(1)
THEN
NOT DB_AMP_Plus_CampGuardOff(1);

// backstop for base spawn paths, delayed so every record of the spawn exists before it is dropped
IF
DB_AMP_MirrorSupport_Spawned((CHARACTER)_Mob, _)
AND
QRY_AMP_Plus_CampGuard()
AND
NOT DB_Players(_Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Plus_CampSweep", 500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_MirrorChase_LastAttacker(_Mob, (CHARACTER)_Att)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_MirrorChase_LastAttacker(_Mob, _Att);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_Plus_Chase(_Mob, (GUIDSTRING)_T)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_Plus_Chase(_Mob, _T);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_Plus_ChaseActive(_Mob, (GUIDSTRING)_T)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_Plus_ChaseActive(_Mob, _T);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_MirrorChase_Mob(_Mob, (GUIDSTRING)_T)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_MirrorChase_Mob(_Mob, _T);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_Plus_UltraMob(_Mob, (CHARACTER)_Boss)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_Plus_UltraMob(_Mob, _Boss);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_CampSweep")
AND
DB_AMP_MirrorSupport_Spawned(_Mob, (CHARACTER)_Boss)
AND
IsDead(_Mob, 0)
THEN
NOT DB_AMP_MirrorSupport_Spawned(_Mob, _Boss);
SetOnStage(_Mob, 0);
RequestDeleteTemporary(_Mob);

EXITSECTION
ENDEXITSECTION
