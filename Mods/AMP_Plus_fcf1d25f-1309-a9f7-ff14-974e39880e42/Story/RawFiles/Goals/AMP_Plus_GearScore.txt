PROC_AMP_Plus_GS_CalcShards((ITEM)_Item)

THEN
NOT DB_AMP_Plus_GS_ShardAcc(_Item, _Old);

PROC
PROC_AMP_Plus_GS_CalcShards((ITEM)_Item)
THEN
DB_AMP_Plus_GS_ShardAcc(_Item, 0);

// stacking shards carry both rows, so count them once
PROC
PROC_AMP_Plus_GS_CalcShards((ITEM)_Item)
AND
DB_AMP_Shard_RecoveryMap((STRING)_Marker, (ITEMROOT)_Template)
AND
HasActiveStatus(_Item, _Marker, 1)
AND
NOT DB_AMP_Shard_Count(_Item, _Marker, _)
AND
DB_ToT_ShardReg(_Template, (STRING)_Tier)
AND
DB_AMP_Plus_GS_ShardW(_Tier, (INTEGER)_W)
AND
DB_AMP_Plus_GS_ShardAcc(_Item, (INTEGER)_Sum)
AND
IntegerSum(_Sum, _W, (INTEGER)_New)
THEN
NOT DB_AMP_Plus_GS_ShardAcc(_Item, _Sum);
DB_AMP_Plus_GS_ShardAcc(_Item, _New);

PROC
PROC_AMP_Plus_GS_CalcShards((ITEM)_Item)
AND
DB_AMP_Shard_Count(_Item, (STRING)_Marker, (INTEGER)_Count)
AND
DB_AMP_Shard_RecoveryMap(_Marker, (ITEMROOT)_Template)
AND
DB_ToT_ShardReg(_Template, (STRING)_Tier)
AND
DB_AMP_Plus_GS_ShardW(_Tier, (INTEGER)_W)
AND
IntegerProduct(_W, _Count, (INTEGER)_Add)
AND
DB_AMP_Plus_GS_ShardAcc(_Item, (INTEGER)_Sum)
AND
IntegerSum(_Sum, _Add, (INTEGER)_New)
THEN
NOT DB_AMP_Plus_GS_ShardAcc(_Item, _Sum);
DB_AMP_Plus_GS_ShardAcc(_Item, _New);

PROC_AMP_Plus_GS_CalcShards((ITEM)_Item)
AND
DB_AMP_Plus_GS_ShardAcc(_Item, (INTEGER)_Sum)
THEN

PROC
PROC_AMP_Plus_GS_Party((GUIDSTRING)_CombatGuid)
AND
CombatGetInvolvedPartyMembersCount(_CombatGuid, (INTEGER)_N)
AND
_N > 0
THEN
DB_AMP_Plus_GS_PartyAcc(_CombatGuid, 0);
PROC_AMP_Plus_GS_PartyIter(_CombatGuid, _N, 1);

PROC
PROC_AMP_Plus_GS_PartyIter((GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx > _Total
THEN
PROC_AMP_Plus_GS_PartyStore(_CombatGuid);

PROC
PROC_AMP_Plus_GS_PartyIter((GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatGuid, _Idx, (CHARACTER)_Member)
AND
IsSummon(_Member, 0)
AND
IntegerSum(_Idx, 1, (INTEGER)_Next)
THEN
PROC_AMP_Plus_GS_Personal(_Member);
PROC_AMP_Plus_GS_AddPeak(_CombatGuid, _Member);
PROC_AMP_Plus_GS_PartyIter(_CombatGuid, _Total, _Next);

PROC
PROC_AMP_Plus_GS_PartyIter((GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatGuid, _Idx, (CHARACTER)_Member)
AND
IsSummon(_Member, 1)
AND
IntegerSum(_Idx, 1, (INTEGER)_Next)
THEN
PROC_AMP_Plus_GS_PartyIter(_CombatGuid, _Total, _Next);

PROC

AND
DB_AMP_Plus_GS_Peak(_Member, (INTEGER)_Peak)
AND
DB_AMP_Plus_GS_ShardSum(_Member, (INTEGER)_Shards)

PROC
PROC_AMP_Plus_GS_PartyStore((GUIDSTRING)_CombatGuid)

NOT DB_AMP_Plus_GS_Party(_CombatGuid, _)
THEN

PROC_AMP_Plus_GS_SetBonus(_CombatGuid, _New);
PROC_AMP_Plus_GS_Log(_CombatGuid, _New);

PROC
PROC_AMP_Plus_GS_PartyStore((GUIDSTRING)_CombatGuid)
AND
DB_AMP_Plus_GS_PartyAcc(_CombatGuid, (INTEGER)_New)
AND
DB_AMP_Plus_GS_Party(_CombatGuid, (INTEGER)_Old)
AND
_New > _Old
THEN
NOT DB_AMP_Plus_GS_Party(_CombatGuid, _Old);
DB_AMP_Plus_GS_Party(_CombatGuid, _New);
PROC_AMP_Plus_GS_SetBonus(_CombatGuid, _New);
PROC_AMP_Plus_GS_Log(_CombatGuid, _New);

PROC
PROC_AMP_Plus_GS_Log((GUIDSTRING)_CombatGuid, (INTEGER)_Score)
AND

AND
Concatenate("AMP_Plus gear score: ", _ScoreStr, (STRING)_Line)
THEN
DebugText(_CombatGuid, _Line);
