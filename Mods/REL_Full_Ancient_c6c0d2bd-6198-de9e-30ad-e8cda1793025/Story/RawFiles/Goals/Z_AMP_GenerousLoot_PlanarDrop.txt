Version 1
SubGoalCombiner SGC_AND
INITSECTION
KBSECTION

// planar coin drop, gated by the generous-loot toggle. low rate, once per fight,
// scaled to host level.

IF
Died((CHARACTER)_Creature)
AND
IsPlayer(_Creature, 0)
AND
DB_AMP_GenerousLoot_Enabled(1)
AND
GetHostCharacter((CHARACTER)_Host)
AND
CombatGetGuidFor(_Host, (GUIDSTRING)_CombatGuid)
AND
NOT DB_AMP_PlanarDrop_Done(_CombatGuid)
AND
GetLevel(_Host, (INTEGER)_Lvl)
AND
Random(100, (INTEGER)_Roll)
AND
_Roll < 5
THEN
DB_AMP_PlanarDrop_Done(_CombatGuid);
PROC_AMP_PlanarDrop_Grant(_Creature, _Lvl);

// level buckets -> matching planar coin into the corpse
PROC
PROC_AMP_PlanarDrop_Grant((CHARACTER)_Creature, (INTEGER)_Lvl)
AND
_Lvl <= 4
THEN
TemplateAddTo(AMP_FeyCoin_c304469f-6132-4304-a5a2-882a2820be3b, _Creature, 1, 0);

PROC
PROC_AMP_PlanarDrop_Grant((CHARACTER)_Creature, (INTEGER)_Lvl)
AND
_Lvl >= 5
AND
_Lvl <= 8
THEN
TemplateAddTo(AMP_ShadowfellCoin_67131136-f11d-4b27-9017-23162629310d, _Creature, 1, 0);

PROC
PROC_AMP_PlanarDrop_Grant((CHARACTER)_Creature, (INTEGER)_Lvl)
AND
_Lvl >= 9
AND
_Lvl <= 12
THEN
TemplateAddTo(AMP_AstralCoin_6696e8d1-d87f-401d-a4b3-ce2966ea5b80, _Creature, 1, 0);

PROC
PROC_AMP_PlanarDrop_Grant((CHARACTER)_Creature, (INTEGER)_Lvl)
AND
_Lvl >= 13
THEN
TemplateAddTo(AMP_CelestialCoin_4dbb1100-ae2a-4107-b69a-099285172cd1, _Creature, 1, 0);

// release the per-fight lock at combat end
IF
CombatEnded((GUIDSTRING)_CombatGuid)
AND
DB_AMP_PlanarDrop_Done(_CombatGuid)
THEN
NOT DB_AMP_PlanarDrop_Done(_CombatGuid);

EXITSECTION
ENDEXITSECTION
