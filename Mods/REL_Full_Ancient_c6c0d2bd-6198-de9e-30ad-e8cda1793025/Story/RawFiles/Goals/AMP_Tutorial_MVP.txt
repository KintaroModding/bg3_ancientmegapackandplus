THEN

AND

AND
QRY_AMP_ShouldShowHint(2)
THEN
OpenMessageBox(_Player, "AMP_Tutorial_Salvage_Station");

AND
NOT DB_AMP_Hints_Mode(0)

AND
QRY_AMP_ShouldShowHint(2)
THEN
