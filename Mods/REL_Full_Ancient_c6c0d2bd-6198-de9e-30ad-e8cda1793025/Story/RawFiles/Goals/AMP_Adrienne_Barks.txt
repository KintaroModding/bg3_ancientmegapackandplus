Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// adrienne combat barks (mirror support pool)

IF
EnteredCombat((CHARACTER)_Mob, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_03_3cfb7f5b-8ba4-4e0a-905d-9d19bf84558b)
AND
NOT DB_AMP_Adrienne_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Adrienne_Greeting_78d75ced-1ad6-40af-9109-ae7582b32674, _Mob)
THEN
DB_AMP_Adrienne_Greeted(_Mob);
DB_AMP_Adrienne_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Adrienne_BarkUnlock", 5000);

IF
TurnStarted((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_03_3cfb7f5b-8ba4-4e0a-905d-9d19bf84558b)
AND
NOT DB_AMP_Adrienne_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Adrienne_Greeting_78d75ced-1ad6-40af-9109-ae7582b32674, _Mob)
THEN
DB_AMP_Adrienne_Greeted(_Mob);
DB_AMP_Adrienne_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Adrienne_BarkUnlock", 5000);

IF
UsingSpell((CHARACTER)_Mob, _, _, _, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_03_3cfb7f5b-8ba4-4e0a-905d-9d19bf84558b)
AND
IsDead(_Mob, 0)
AND
NOT DB_AMP_Adrienne_BarkLock(_Mob)
AND
Random(3, (INTEGER)_Roll)
AND
_Roll == 0
THEN
DB_AMP_Adrienne_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Adrienne_BarkDelay", 1500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Adrienne_BarkDelay")
AND
DB_AMP_Adrienne_BarkLock(_Mob)
AND
IsDead(_Mob, 0)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Adrienne_SpellCast_5a98a6f1-7346-4e10-8d23-f2d10740fb59, _Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Adrienne_BarkUnlock", 10000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Adrienne_BarkUnlock")
AND
DB_AMP_Adrienne_BarkLock(_Mob)
THEN
NOT DB_AMP_Adrienne_BarkLock(_Mob);

IF
HitpointsChanged((CHARACTER)_Mob, (REAL)_Pct)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_03_3cfb7f5b-8ba4-4e0a-905d-9d19bf84558b)
AND
_Pct <= 40.0
AND
NOT DB_AMP_Adrienne_LowHP(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Adrienne_LowHP_f4782680-92d8-43ac-afd0-639a2b717400, _Mob)
THEN
DB_AMP_Adrienne_LowHP(_Mob);

IF
Died((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_03_3cfb7f5b-8ba4-4e0a-905d-9d19bf84558b)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Adrienne_Death_e4608d6e-4136-48a1-9550-b5fbf5d80a52, _Mob)
THEN
DB_NOOP(1);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Adrienne_Greeted(_Mob)
THEN
NOT DB_AMP_Adrienne_Greeted(_Mob);
DB_AMP_Adrienne_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Adrienne_BarkUnlock", 5000);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Adrienne_BarkLock(_Mob)
THEN
NOT DB_AMP_Adrienne_BarkLock(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Adrienne_LowHP(_Mob)
THEN
NOT DB_AMP_Adrienne_LowHP(_Mob);

EXITSECTION

ENDEXITSECTION
