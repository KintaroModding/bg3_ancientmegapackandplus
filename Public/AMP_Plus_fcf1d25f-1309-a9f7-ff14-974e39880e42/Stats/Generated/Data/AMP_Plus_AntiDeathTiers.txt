// Hard is the base carrier. each downed status clears its own charge, so a boss can't rise twice off one grant

type "StatusData"

using "AMP_BOSS_ANTIDEATH_DOWNED"
data "OnApplyFunctors" "RemoveStatus(AMP_Plus_ANTIDEATH_CHARGE_STD);RegainHitPoints(MaxHP/4,Guaranteed);ApplyStatus(ACTION_SURGE,100,1);ApplyStatus(AMP_BOSS_POSSESSED,100,-1)"
