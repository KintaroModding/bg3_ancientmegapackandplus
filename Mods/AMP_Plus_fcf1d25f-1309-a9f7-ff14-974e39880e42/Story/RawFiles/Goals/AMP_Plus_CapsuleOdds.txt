PROC
PROC_AMP_Plus_CapOdds_Bump((STRING)_Kind)
AND
DB_AMP_Plus_CapOdds_Heat(_Kind, (INTEGER)_H)
AND
DB_AMP_Plus_CapOdds_HeatCap((INTEGER)_Cap)
AND
_H < _Cap
AND
IntegerSum(_H, 1, (INTEGER)_Next)
THEN
NOT DB_AMP_Plus_CapOdds_Heat(_Kind, _H);
DB_AMP_Plus_CapOdds_Heat(_Kind, _Next);

THEN
DB_AMP_Plus_CapOdds_Heat(_Kind, 1);

PROC_AMP_Plus_CapOdds_Bump((STRING)_Kind)
THEN
PROC_AMP_Plus_CapOdds_Arm(_Kind);

// one recorded carrier, the host follows whoever is controlled and would grow a clock on every character
PROC
PROC_AMP_Plus_CapOdds_Arm((STRING)_Kind)
AND
DB_AMP_Plus_CapOdds_Carrier(_Kind, (CHARACTER)_Char)
AND
DB_AMP_Plus_CapOdds_Clock(_Kind, (STRING)_Status)
AND
NOT DB_Players(_Char)
THEN
NOT DB_AMP_Plus_CapOdds_Carrier(_Kind, _Char);
RemoveStatus(_Char, _Status);

PROC
PROC_AMP_Plus_CapOdds_Arm((STRING)_Kind)
AND
DB_AMP_Plus_CapOdds_Carrier(_Kind, (CHARACTER)_Char)
AND
DB_AMP_Plus_CapOdds_Clock(_Kind, (STRING)_Status)
AND
HasAppliedStatus(_Char, _Status, 0)
THEN
NOT DB_AMP_Plus_CapOdds_Carrier(_Kind, _Char);

PROC
PROC_AMP_Plus_CapOdds_Arm((STRING)_Kind)
AND
NOT DB_AMP_Plus_CapOdds_Carrier(_Kind, _)
AND
DB_AMP_Plus_CapOdds_Clock(_Kind, (STRING)_Status)
AND
DB_AMP_Plus_CapOdds_Heat(_Kind, (INTEGER)_H)
AND
_H > 0
AND
GetHostCharacter((CHARACTER)_Host)
THEN
DB_AMP_Plus_CapOdds_Carrier(_Kind, _Host);
ApplyStatus(_Host, _Status, 90.0, 1, NULL_00000000-0000-0000-0000-000000000000);

// re-arm waits, the status still reads as applied inside its own removal event
IF
StatusRemoved((CHARACTER)_Char, (STRING)_Status, _, _)
AND
DB_AMP_Plus_CapOdds_Clock((STRING)_Kind, _Status)
AND
DB_AMP_Plus_CapOdds_Carrier(_Kind, _Char)
THEN
NOT DB_AMP_Plus_CapOdds_Carrier(_Kind, _Char);
PROC_AMP_Plus_CapOdds_Cool(_Kind);
TimerLaunch("AMP_Plus_CapOdds_Rearm", 200);
TimerLaunch("AMP_Plus_CapOdds_Stamp", 50);

PROC_AMP_Plus_CapOdds_Cool((STRING)_Kind)

DB_AMP_Plus_CapOdds_Heat(_Kind, (INTEGER)_H)

_H > 0
AND
IntegerSubtract(_H, 1, (INTEGER)_Next)
THEN
NOT DB_AMP_Plus_CapOdds_Heat(_Kind, _H);
DB_AMP_Plus_CapOdds_Heat(_Kind, _Next);

TimerFinished("AMP_Plus_CapOdds_Rearm")
THEN
PROC_AMP_Plus_CapOdds_Arm("SALVAGE");
PROC_AMP_Plus_CapOdds_Arm("DUST");
