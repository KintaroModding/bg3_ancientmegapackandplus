KBSECTION
// first use - show tutorial + mark player ready

IF
UseStarted((CHARACTER)_Player, (ITEM)_Button)
AND
DB_AMP_CoinVendor_Button(_Button, _, _)
AND
NOT DB_AMP_CoinVendor_Ready(_Player)
AND
DB_Players((CHARACTER)_Player)
THEN
DB_AMP_CoinVendor_Ready(_Player);
PROC_AMP_CoinVendor_ShowTutorial(_Player);

// mode 0: always show, no flag tracking
PROC
PROC_AMP_CoinVendor_ShowTutorial((CHARACTER)_Player)
AND
DB_AMP_Hints_Mode(0)
AND
QRY_AMP_ShouldShowHint(2)
THEN
OpenMessageBox(_Player, "AMP_Settings016");

// modes 1/3/4: show once, track with flag
PROC
PROC_AMP_CoinVendor_ShowTutorial((CHARACTER)_Player)
AND
NOT DB_AMP_Hints_Mode(0)
AND
NOT DB_AMP_Tutorial_Shown(_Player, "CoinVendorFirst")
AND
QRY_AMP_ShouldShowHint(2)
THEN
OpenMessageBox(_Player, "AMP_Settings016");
DB_AMP_Tutorial_Shown((CHARACTER)_Player, "CoinVendorFirst");

// button handler

// success - enough gold (personal, not magic pockets)
IF
UseStarted((CHARACTER)_Player, (ITEM)_Button)
AND
DB_AMP_CoinVendor_Button(_Button, _CoinTemplate, _Price)
AND
DB_AMP_CoinVendor_Ready(_Player)
AND
GetGold(_Player, _Gold)
AND
_Gold >= _Price
AND
IntegerProduct(_Price, -1, _MinusPrice)
THEN
AddGold(_Player, _MinusPrice);
TemplateAddTo(_CoinTemplate, _Player, 1, 1);
PlaySound(_Player, "CrSpell_Cast_MagicClub");

// not enough gold
IF
UseStarted((CHARACTER)_Player, (ITEM)_Button)
AND
DB_AMP_CoinVendor_Button(_Button, _, _Price)
AND
DB_AMP_CoinVendor_Ready(_Player)
AND
GetGold(_Player, _Gold)
AND
_Gold < _Price
THEN
PlaySound(_Player, "UI_HUD_Combine_Error");
