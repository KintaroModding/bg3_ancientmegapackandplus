// activate on use start
IF
UseStarted((CHARACTER)_Player, (ITEM)_Station)
AND
IsTagged((ITEM)_Station, (TAG)AMP_STATION_2_63e37ef0-064c-4065-a939-dc9724b9c85f, 1)
AND
DB_Players((CHARACTER)_Player)
THEN
RemoveStatus((ITEM)_Station, "ARCANIZER_STATION_ATTENTION_VFX");

IF

AND
IsTagged((ITEM)_Station, (TAG)AMP_STATION_1_e39240da-bf08-4793-bd1f-609c93e4fa56, 1)
AND
IsTagged((ITEM)_Station, (TAG)AMP_STATION_2_63e37ef0-064c-4065-a939-dc9724b9c85f, 0)
AND

AND

AND
IsTagged((ITEM)_Station, (TAG)AMP_STATION_2_63e37ef0-064c-4065-a939-dc9724b9c85f, 0)

THEN
