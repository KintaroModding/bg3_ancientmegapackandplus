Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION

// cancels the base spawn-time strip for boss-tied mobs, self-registered pursuers keep it
IF
StatusApplied((CHARACTER)_Mob, "AMP_MIRROR_SUPPORT_FLAG", _, _)
AND
DB_AMP_Plus_UltraMob(_Mob, (CHARACTER)_Boss)
AND
NOT _Mob == _Boss
THEN
RealtimeObjectTimerCancel(_Mob, "AMP_MirrorStripDelay");

// strip while the mobs are alive and iterable, before the boss death removes them
IF
StatusApplied((CHARACTER)_Boss, "AMP_PLUS_MIRROR_STRIP_SIGNAL", _, _)
AND
DB_AMP_Plus_UltraMob((CHARACTER)_Mob, _Boss)
AND
NOT _Mob == _Boss
AND
IsDead(_Mob, 0)
THEN
IterateInventory(_Mob, "AMP_Plus_MirrorStrip", "");

IF
StatusApplied((CHARACTER)_Boss, "AMP_PLUS_MIRROR_STRIP_SIGNAL", _, _)
THEN
IterateInventory(_Boss, "AMP_Plus_MirrorStrip", "");

// delete on the corpse before looting. second pass for an unreadable death-frame inventory
IF
Died((CHARACTER)_Mob)
AND
HasActiveStatus(_Mob, "AMP_MIRROR_SUPPORT_FLAG", 1)
THEN
IterateInventory(_Mob, "AMP_Plus_MirrorStrip", "");
RealtimeObjectTimerLaunch(_Mob, "AMP_Plus_MirrorStripCorpse", 1000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Plus_MirrorStripCorpse")
THEN
IterateInventory(_Mob, "AMP_Plus_MirrorStrip", "");

// vanilla boss inventories hold unequipped quest items, so drinkables and throwables only
IF
EntityEvent((ITEM)_Item, "AMP_Plus_MirrorStrip")
AND
IsEquipped(_Item, 0)
AND
IsConsumable(_Item, 1)
AND
NOT IsStoryItem(_Item, 1)
THEN
RequestDelete(_Item);

EXITSECTION
ENDEXITSECTION
