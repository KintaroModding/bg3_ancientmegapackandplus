// lever used - hide tree (cooldown gated)
IF
UseStarted(_, (ITEM)_Lever)
AND
DB_AMP_LeverTree(_Lever, _Tree)
AND
NOT DB_AMP_LeverCooldown(_Lever)
AND
DB_AMP_TreeVisible(_Tree, 1)
THEN
DebugText(_Lever, "HIDING TREE");
DB_AMP_LeverCooldown(_Lever);
ObjectTimerLaunch(_Lever, "AMP_LeverCooldown", 500, 0);
NOT DB_AMP_TreeVisible(_Tree, 1);
DB_AMP_TreeVisible(_Tree, 0);
SetOnStage(_Tree, 0);

// lever used - show tree (cooldown gated)
IF
UseStarted(_, (ITEM)_Lever)
AND
DB_AMP_LeverTree(_Lever, _Tree)
AND
NOT DB_AMP_LeverCooldown(_Lever)
AND
DB_AMP_TreeVisible(_Tree, 0)
THEN
DebugText(_Lever, "SHOWING TREE");
DB_AMP_LeverCooldown(_Lever);
ObjectTimerLaunch(_Lever, "AMP_LeverCooldown", 500, 0);
NOT DB_AMP_TreeVisible(_Tree, 0);
DB_AMP_TreeVisible(_Tree, 1);
SetOnStage(_Tree, 1);

// clear cooldown when timer ends
IF
ObjectTimerFinished((ITEM)_Lever, "AMP_LeverCooldown")
AND
DB_AMP_LeverCooldown(_Lever)
THEN
NOT DB_AMP_LeverCooldown(_Lever);
