// chaos equip: destroy a random equipped item

// status -> roll
IF
StatusApplied((CHARACTER)_Char, "AMP_BOOST18_CHAOS_EQUIP", _, _)
THEN
RemoveStatus(_Char, "AMP_BOOST18_CHAOS_EQUIP");
PROC_AMP_Boost18_ChanceRoll(_Char);

// 20% chance: random(5)==0, else the and fails and nothing happens
PROC
PROC_AMP_Boost18_ChanceRoll((CHARACTER)_Char)
AND
Random(5, (INTEGER)_Chance)
AND
_Chance == 0
THEN
DB_AMP_Boost18_SlotCount(_Char, 0);
PROC_AMP_Boost18_CollectSlots(_Char);
PROC_AMP_Boost18_RollAndDestroy(_Char);

// collect equipped items into an indexed db (one proc per slot, runs in sequence)

PROC
PROC_AMP_Boost18_CollectSlots((CHARACTER)_Char)
AND
GetEquippedItem(_Char, "Helmet", (ITEM)_Item)
AND
DB_AMP_Boost18_SlotCount(_Char, (INTEGER)_N)
AND
IntegerSum(_N, 1, (INTEGER)_Next)
THEN
NOT DB_AMP_Boost18_SlotCount(_Char, _N);
DB_AMP_Boost18_SlotCount(_Char, _Next);
DB_AMP_Boost18_Equipped(_Char, _N, _Item);

// pick a random collected item and destroy it
PROC
PROC_AMP_Boost18_RollAndDestroy((CHARACTER)_Char)
AND
DB_AMP_Boost18_SlotCount(_Char, (INTEGER)_Count)
AND
_Count > 0
AND
Random(_Count, (INTEGER)_Roll)
AND
DB_AMP_Boost18_Equipped(_Char, _Roll, (ITEM)_Item)
THEN
RequestDelete(_Item);
PROC_AMP_Boost18_Cleanup(_Char);

// nothing equipped (count 0): just cleanup
PROC
PROC_AMP_Boost18_RollAndDestroy((CHARACTER)_Char)
AND
DB_AMP_Boost18_SlotCount(_Char, 0)
THEN
NOT DB_AMP_Boost18_SlotCount(_Char, 0);

// cleanup db entries
PROC
PROC_AMP_Boost18_Cleanup((CHARACTER)_Char)
AND
DB_AMP_Boost18_Equipped(_Char, (INTEGER)_N, (ITEM)_Item)
THEN
NOT DB_AMP_Boost18_Equipped(_Char, _N, _Item);

PROC
PROC_AMP_Boost18_Cleanup((CHARACTER)_Char)
AND
DB_AMP_Boost18_SlotCount(_Char, (INTEGER)_N)
THEN
NOT DB_AMP_Boost18_SlotCount(_Char, _N);
