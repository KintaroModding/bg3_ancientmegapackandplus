INITSECTION

KBSECTION

QRY
QRY_AMP_BossRewardsEnabled()
AND
GetHostCharacter((CHARACTER)_Host)
AND
HasActiveStatus(_Host, "AMP_BOSS_REWARDS_DISABLED", 0)
THEN
DB_NOOP(1);

// deferred reward delivery

// in combat -> queue, deliver after combat ends
PROC
PROC_AMP_BossRewardGrant((CHARACTER)_Player, (STRING)_RewardID)
AND
IsInCombat(_Player, 1)
THEN
DB_AMP_BossReward_Pending(_Player, _RewardID);

// not in combat -> deliver now
PROC
PROC_AMP_BossRewardGrant((CHARACTER)_Player, (STRING)_RewardID)
AND
IsInCombat(_Player, 0)
THEN
QuestUpdate(_Player, "HIDDEN_WLD_Boosters", _RewardID);

// flush queued rewards on combat end
IF
CombatEnded(_CombatGuid)
AND
DB_AMP_BossReward_Pending(_Player, _RewardID)
THEN
QuestUpdate(_Player, "HIDDEN_WLD_Boosters", _RewardID);
NOT DB_AMP_BossReward_Pending(_Player, _RewardID);
