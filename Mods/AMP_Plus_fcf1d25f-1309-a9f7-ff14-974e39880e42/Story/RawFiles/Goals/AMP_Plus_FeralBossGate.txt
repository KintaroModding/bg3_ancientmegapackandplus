KBSECTION

// decided once at first grade, a re-tier would heal the unit

IF
EnteredCombat((CHARACTER)_Boss, (GUIDSTRING)_CombatID)
AND
IsBoss(_Boss, 1)
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsEnemy(_Boss, _Host, 1)
AND
NOT DB_AMP_Plus_FeralBossCombat(_CombatID)
THEN
DB_AMP_Plus_FeralBossCombat(_CombatID);

IF
DB_AMP_DiffV3_Tracked((CHARACTER)_C, _)
AND
NOT DB_AMP_Plus_FeralMult(_C, _)
AND
IsBoss(_C, 1)
THEN
DB_AMP_Plus_FeralMult(_C, 0);

IF
DB_AMP_DiffV3_Tracked((CHARACTER)_C, _)
AND
NOT DB_AMP_Plus_FeralMult(_C, _)
AND
IsBoss(_C, 0)
AND
IsInCombat(_C, 1)
AND
CombatGetGuidFor(_C, (GUIDSTRING)_CombatID)
AND
DB_AMP_Plus_FeralBossCombat(_CombatID)
THEN
DB_AMP_Plus_FeralMult(_C, 0);

IF
DB_AMP_DiffV3_Tracked((CHARACTER)_C, _)
AND
NOT DB_AMP_Plus_FeralMult(_C, _)
AND
IsBoss(_C, 0)
AND
IsInCombat(_C, 1)
AND
CombatGetGuidFor(_C, (GUIDSTRING)_CombatID)
AND
NOT DB_AMP_Plus_FeralBossCombat(_CombatID)
THEN
DB_AMP_Plus_FeralMult(_C, 1);

IF
DB_AMP_DiffV3_Tracked((CHARACTER)_C, _)
AND
NOT DB_AMP_Plus_FeralMult(_C, _)
AND
IsBoss(_C, 0)
AND
IsInCombat(_C, 0)
THEN
DB_AMP_Plus_FeralMult(_C, 1);

IF
CombatEnded((GUIDSTRING)_CombatID)
AND
DB_AMP_Plus_FeralBossCombat(_CombatID)
THEN
NOT DB_AMP_Plus_FeralBossCombat(_CombatID);

EXITSECTION
ENDEXITSECTION
