//REGION Snake Court (DEN_SnakeCourt, Act 1) - vanilla debug resolvers in Act1_DEN_ShadowDruid.txt

IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Plus_StorySkip_SnakeCourt", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
DB_CurrentLevel(_Level)
AND
_Level != "WLD_Main_A"
THEN
OpenMessageBox(_Caster, "AMP_Plus_StorySkip_WrongAct");

IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Plus_StorySkip_SnakeCourt", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
DB_CurrentLevel("WLD_Main_A")
AND
GetFlag(DEN_ShadowDruid_State_SnakesCourtDone_cf58dd01-e058-42b8-b1f1-976250296ce6, NULL_00000000-0000-0000-0000-000000000000, 1)
THEN
OpenMessageBox(_Caster, "AMP_Plus_StorySkip_Done");

IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Plus_StorySkip_SnakeCourt", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
DB_CurrentLevel("WLD_Main_A")
AND
GetFlag(DEN_ShadowDruid_State_SnakesCourtDone_cf58dd01-e058-42b8-b1f1-976250296ce6, NULL_00000000-0000-0000-0000-000000000000, 0)
THEN
OpenMessageBoxYesNo(_Caster, "AMP_Plus_StorySkip_SnakeCourt_AskFree");

// reunion step runs on a timer so the court journal entries settle first
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Plus_StorySkip_SnakeCourt_AskFree", 1)
AND
GetFlag(DEN_ShadowDruid_State_SnakesCourtDone_cf58dd01-e058-42b8-b1f1-976250296ce6, NULL_00000000-0000-0000-0000-000000000000, 0)
THEN
SetFlag(DEN_ShadowDruid_Debug_ResolveKidAlive_dfcf720f-16ce-b7e6-bfeb-d111b90b2f0f, _Host);
TimerLaunch("AMP_Plus_StorySkip_SnakeCourt_Thankful", 3000);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Plus_StorySkip_SnakeCourt_AskFree", 0)
AND
GetFlag(DEN_ShadowDruid_State_SnakesCourtDone_cf58dd01-e058-42b8-b1f1-976250296ce6, NULL_00000000-0000-0000-0000-000000000000, 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Plus_StorySkip_SnakeCourt_AskDie");

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Plus_StorySkip_SnakeCourt_AskDie", 1)
AND
GetFlag(DEN_ShadowDruid_State_SnakesCourtDone_cf58dd01-e058-42b8-b1f1-976250296ce6, NULL_00000000-0000-0000-0000-000000000000, 0)
THEN
SetFlag(DEN_ShadowDruid_Debug_ResolveKidDie_76076244-4ddf-f7e1-c288-6572d72bf3c1, _Host);
TimerLaunch("AMP_Plus_StorySkip_SnakeCourt_Devastated", 3000);

// engine grants the step XP itself
IF
TimerFinished("AMP_Plus_StorySkip_SnakeCourt_Thankful")
AND
GetHostCharacter(_Host)
THEN
QuestUpdate(_Host, "DEN_SnakeCourt", "ParentThankful");

IF
TimerFinished("AMP_Plus_StorySkip_SnakeCourt_Devastated")
AND
GetHostCharacter(_Host)
THEN
QuestUpdate(_Host, "DEN_SnakeCourt", "ParentDevastated");

PROC

THEN
DB_AMP_Plus_Act1Finale_Done(1);
OpenMessageBox(_Host, "AMP_Plus_Act1Finale_Running");
TimerLaunch("AMP_Plus_Act1Finale_W1", 2000);
TimerLaunch("AMP_Plus_Act1Finale_W2", 5000);
TimerLaunch("AMP_Plus_Act1Finale_W3", 8000);
TimerLaunch("AMP_Plus_Act1Finale_W4", 11000);
TimerLaunch("AMP_Plus_Act1Finale_W5", 14000);
TimerLaunch("AMP_Plus_Act1Finale_W6", 17000);
TimerLaunch("AMP_Plus_Act1Finale_W7", 20000);

IF
TimerFinished("AMP_Plus_Act1Finale_W1")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Good")
THEN
PROC_DEN_DEBUG_TieflingRefugees_KillGoblinLeadership(_Host);
SetFlag(GLO_Halsin_Debug_LeadersDead_4037ac74-9e75-4339-abbb-9b7e83c70bc9, _Host);
PROC_DEN_TieflingRefugees_SetGoblinHuntVictoryState();
PROC_State_Progress(S_DEN_RangerDen_SUB_50062397-bf9c-4765-9cbc-e40b5148f211, "DEN", "DEN_State_GoblinHuntVictory");

IF
TimerFinished("AMP_Plus_Act1Finale_W1")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Evil")
THEN
PROC_DEN_RaidingParty_DebugResolveConfrontation();
SetFlag(DEN_AttackOnDen_Debug_Start_10b6026f-b246-7f88-aa40-a653fe08d861, NULL_00000000-0000-0000-0000-000000000000, 0);

IF
TimerFinished("AMP_Plus_Act1Finale_W2")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Good")
THEN
PROC_DEN_TieflingRefugees_DebugLeave();
PROC_State_Progress(S_DEN_RangerDen_SUB_50062397-bf9c-4765-9cbc-e40b5148f211, "DEN", "DEN_State_TieflingsGone");

TimerFinished("AMP_Plus_Act1Finale_W2")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Good")
AND
DB_AMP_Plus_Act1Finale_Flag("Halsin")
THEN

AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Good")
AND
NOT DB_AMP_Plus_Act1Finale_Flag("Kagha")

TimerFinished("AMP_Plus_Act1Finale_W2")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone("Evil")
THEN

TimerFinished("AMP_Plus_Act1Finale_W3")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Flag("Sweep")
AND
DB_AMP_Plus_Act1Finale_Tone(_Tone)
THEN

PROC

AND
NOT DB_AMP_Plus_Act1Finale_SpareGroup(_Tone, _Group)
AND
NOT DB_PermaDefeated((CHARACTER)_C)

AND

AND
DB_AMP_Plus_Act1Finale_Tone(_Tone)
THEN

TimerFinished("AMP_Plus_Act1Finale_W5")
AND
GetHostCharacter(_Host)
AND
DB_AMP_Plus_Act1Finale_Tone(_Tone)
THEN
PROC_AMP_Plus_Act1Finale_QuestSweep(_Host, _Tone, 2);

DB_AMP_Plus_Act1Finale_Flag("XP")

THEN
