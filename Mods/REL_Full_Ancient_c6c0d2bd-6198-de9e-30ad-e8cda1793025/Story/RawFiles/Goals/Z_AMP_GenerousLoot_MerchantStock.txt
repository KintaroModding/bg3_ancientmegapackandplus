// auto-extract only ejects to a player owner, so cubes stay sellable containers in a merchant

// act 1 -> tier-1 cubes
IF
LevelGameplayStarted(_, _)
AND
DB_AMP_GenerousLoot_Enabled(1)
AND
DB_AMP_TraderAct((CHARACTER)_Trader, 1)
AND
Exists(_Trader, 1)
AND
NOT DB_AMP_GenMerchantStocked(_Trader)
AND
Random(6, (INTEGER)_R)
AND
IntegerSum(_R, 5, (INTEGER)_Count)
THEN
DB_AMP_GenMerchantStocked(_Trader);
TemplateAddTo(AMP_Capsule_Reward_Generic_1_bc44d261-edfb-424e-ab61-98d16899ab7b, _Trader, _Count, 0);
