Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// vasht combat barks (pool boss)

IF
EnteredCombat((CHARACTER)_Mob, _)
AND
GetTemplate(_Mob, AMP_Ysatiy_92506335-12dd-4e4b-848c-f33fd8cc39fa)
AND
NOT DB_AMP_Vasht_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vasht_Greeting_b6597672-80b2-4dfe-bbc2-7b117cb083d1, _Mob)
THEN
DB_AMP_Vasht_Greeted(_Mob);
DB_AMP_Vasht_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vasht_BarkUnlock", 5000);

IF
TurnStarted((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_Ysatiy_92506335-12dd-4e4b-848c-f33fd8cc39fa)
AND
NOT DB_AMP_Vasht_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vasht_Greeting_b6597672-80b2-4dfe-bbc2-7b117cb083d1, _Mob)
THEN
DB_AMP_Vasht_Greeted(_Mob);
DB_AMP_Vasht_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vasht_BarkUnlock", 5000);

IF
UsingSpell((CHARACTER)_Mob, _, _, _, _)
AND
GetTemplate(_Mob, AMP_Ysatiy_92506335-12dd-4e4b-848c-f33fd8cc39fa)
AND
IsDead(_Mob, 0)
AND
NOT DB_AMP_Vasht_BarkLock(_Mob)
AND
Random(3, (INTEGER)_Roll)
AND
_Roll == 0
THEN
DB_AMP_Vasht_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vasht_BarkDelay", 1500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Vasht_BarkDelay")
AND
DB_AMP_Vasht_BarkLock(_Mob)
AND
IsDead(_Mob, 0)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vasht_SpellCast_3e78431b-fa92-4e72-bf70-e068630b3764, _Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Vasht_BarkUnlock", 10000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Vasht_BarkUnlock")
AND
DB_AMP_Vasht_BarkLock(_Mob)
THEN
NOT DB_AMP_Vasht_BarkLock(_Mob);

IF
HitpointsChanged((CHARACTER)_Mob, (REAL)_Pct)
AND
GetTemplate(_Mob, AMP_Ysatiy_92506335-12dd-4e4b-848c-f33fd8cc39fa)
AND
_Pct <= 35.0
AND
NOT DB_AMP_Vasht_LowHP(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vasht_LowHP_25f6ef07-2b2f-48e6-994d-9286c13cdaef, _Mob)
THEN
DB_AMP_Vasht_LowHP(_Mob);

IF
Died((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_Ysatiy_92506335-12dd-4e4b-848c-f33fd8cc39fa)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vasht_Death_bce23fb7-4e5b-4550-ae6f-892d4a597bd5, _Mob)
THEN
DB_NOOP(1);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vasht_Greeted(_Mob)
THEN
NOT DB_AMP_Vasht_Greeted(_Mob);
DB_AMP_Vasht_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vasht_BarkUnlock", 5000);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vasht_BarkLock(_Mob)
THEN
NOT DB_AMP_Vasht_BarkLock(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vasht_LowHP(_Mob)
THEN
NOT DB_AMP_Vasht_LowHP(_Mob);

EXITSECTION

ENDEXITSECTION
