// echo of the litany: cloak wearer heals a stigma-carrying ally -> every other ally
// carrying one of the wearer's stigmas also heals. marker per-tier so it scales with the cloak.

IF
StatusApplied((CHARACTER)_Target, "AMP_ILMATER_LITANY_HEAL_MARKER_T2", (CHARACTER)_Caster, _)
AND
DB_AMP_Ilmater_StigmaLink(_Caster, _OtherTarget, _AnyStigma)
AND
IsDead(_OtherTarget, 0)
AND
_OtherTarget != _Target
THEN
ApplyStatus(_OtherTarget, "AMP_ILMATER_ECHO_HEAL_T2", 0.0, 1, _Caster);
