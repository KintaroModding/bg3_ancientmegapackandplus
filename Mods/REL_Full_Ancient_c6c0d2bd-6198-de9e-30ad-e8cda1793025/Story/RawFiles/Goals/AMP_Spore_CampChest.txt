Version 1
SubGoalCombiner SGC_AND

INITSECTION

KBSECTION

// long rest: run spore infection on each camp chest
IF
LongRestFinished()
AND
DB_Camp_UserCampChest(_, (ITEM)_Chest)
THEN
PROC_AMP_SporeChest_Init(_Chest);
PROC_AMP_SporeChest_RunSources(_Chest);

// reset pass counter
PROC
PROC_AMP_SporeChest_Init((ITEM)_Chest)
AND
DB_AMP_SporeChest_PassNew(_Chest, (INTEGER)_Old)
THEN
NOT DB_AMP_SporeChest_PassNew(_Chest, _Old);

PROC
PROC_AMP_SporeChest_Init((ITEM)_Chest)
THEN
DB_AMP_SporeChest_PassNew(_Chest, 0);

// source 1: spore item sitting in the chest
PROC
PROC_AMP_SporeChest_RunSources((ITEM)_Chest)
AND
TemplateIsInInventory((ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9, _Chest, (INTEGER)_Count)
AND
_Count > 0
THEN
PROC_AMP_SporeChest_RunPass(_Chest);

// source 2: each already-infected item in the chest
PROC
PROC_AMP_SporeChest_RunSources((ITEM)_Chest)
AND
DB_AMP_Spore_ItemInfections((ITEM)_InfItem, (INTEGER)_InfCount)
AND
_InfCount > 0
AND
IsInInventoryOf(_InfItem, _Chest, 1)
THEN
PROC_AMP_SporeChest_RunMulti(_Chest, _InfCount);

// one pass per infection count, capped at 10
PROC
PROC_AMP_SporeChest_RunMulti((ITEM)_Chest, (INTEGER)_Remaining)
AND
_Remaining > 0
AND
DB_AMP_SporeChest_PassNew(_Chest, (INTEGER)_PassCount)
AND
_PassCount < 10
AND
IntegerSubtract(_Remaining, 1, _NextRemaining)
THEN
PROC_AMP_SporeChest_RunPass(_Chest);
PROC_AMP_SporeChest_RunMulti(_Chest, _NextRemaining);

// one pass: walk the chest inventory
PROC
PROC_AMP_SporeChest_RunPass((ITEM)_Chest)
THEN
DB_AMP_SporeChest_Current(_Chest);
IterateInventory(_Chest, "AMP_SporeChest_TryInfect", "AMP_SporeChest_Done");

// try to infect each item, 3% roll
IF
EntityEvent((ITEM)_Item, "AMP_SporeChest_TryInfect")
AND
DB_AMP_SporeChest_Current((ITEM)_Chest)
AND
DB_AMP_SporeChest_PassNew(_Chest, (INTEGER)_PassCount)
AND
_PassCount < 10
AND
NOT GetTemplate(_Item, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)
AND
NOT IsStoryItem(_Item, 1)
AND
Random(1000, (INTEGER)_Roll)
AND
_Roll < 30
AND
IntegerSum(_PassCount, 1, _NewPassCount)
THEN
ApplyStatus(_Item, "AMP_SPORE_INFECTION", -1.0, 1, _Chest);
PROC_AMP_Spore_TrackInfection(_Item);
NOT DB_AMP_SporeChest_PassNew(_Chest, _PassCount);
DB_AMP_SporeChest_PassNew(_Chest, _NewPassCount);

// cleanup after pass
IF
EntityEvent(_, "AMP_SporeChest_Done")
AND
DB_AMP_SporeChest_Current((ITEM)_Chest)
THEN
NOT DB_AMP_SporeChest_Current(_Chest);

EXITSECTION
SysClear("DB_AMP_SporeChest_PassNew", 2);
SysClear("DB_AMP_SporeChest_Current", 1);

ENDEXITSECTION
