// normal mode: roll with diminishing returns
PROC
PROC_AMP_ExchangeCrystal((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType)
AND
DB_AMP_Cheat_Mode(0)
AND
DB_AMP_Altar_LastUser((CHARACTER)_Player)
AND
DB_AMP_Altar_BaseChance(_CrystalType, (INTEGER)_BaseChance)
AND
DB_AMP_Altar_Diminishing(_CrystalType, (INTEGER)_DiminishingRate)
THEN
PROC_AMP_CalculateAndRoll(_Crystal, _Altar, _CrystalType, _Player, _BaseChance, _DiminishingRate);

// get success count for this altar + crystal type
PROC
PROC_AMP_CalculateAndRoll((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_BaseChance, (INTEGER)_DiminishingRate)
AND
DB_AMP_Altar_Successes(_Altar, _CrystalType, (INTEGER)_SuccessCount)
THEN
PROC_AMP_ApplyDiminishing(_Crystal, _Altar, _CrystalType, _Player, _BaseChance, _DiminishingRate, _SuccessCount);

// no prior successes -> 0
PROC
PROC_AMP_CalculateAndRoll((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_BaseChance, (INTEGER)_DiminishingRate)
AND
NOT DB_AMP_Altar_Successes(_Altar, _CrystalType, _)
THEN
PROC_AMP_ApplyDiminishing(_Crystal, _Altar, _CrystalType, _Player, _BaseChance, _DiminishingRate, 0);

// fresh -> base chance
PROC
PROC_AMP_ApplyDiminishing((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_BaseChance, (INTEGER)_DiminishingRate, 0)
THEN
PROC_AMP_RollForSuccess(_Crystal, _Altar, _CrystalType, _Player, _BaseChance);

// prior successes -> apply diminishing
PROC
PROC_AMP_ApplyDiminishing((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_BaseChance, (INTEGER)_DiminishingRate, (INTEGER)_SuccessCount)
AND
_SuccessCount > 0
THEN
PROC_AMP_CalculateMultiplicative(_Crystal, _Altar, _CrystalType, _Player, _BaseChance, _DiminishingRate, _SuccessCount);

// chance = base * ((100-rate)/100)^count. one success
PROC
PROC_AMP_CalculateMultiplicative((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_CurrentChance, (INTEGER)_DiminishingRate, 1)
AND
IntegerSubtract(100, _DiminishingRate, (INTEGER)_Multiplier)
AND
IntegerProduct(_CurrentChance, _Multiplier, (INTEGER)_Product)
AND
IntegerDivide(_Product, 100, (INTEGER)_NewChance)
AND
IntegerMax(_NewChance, 1, (INTEGER)_FinalChance)
THEN
PROC_AMP_RollForSuccess(_Crystal, _Altar, _CrystalType, _Player, _FinalChance);

// multiple successes -> iterate
PROC
PROC_AMP_CalculateMultiplicative((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_CurrentChance, (INTEGER)_DiminishingRate, (INTEGER)_RemainingCount)
AND
_RemainingCount > 1
AND
IntegerSubtract(100, _DiminishingRate, (INTEGER)_Multiplier)
AND
IntegerProduct(_CurrentChance, _Multiplier, (INTEGER)_Product)
AND
IntegerDivide(_Product, 100, (INTEGER)_NewChance)
AND
IntegerSubtract(_RemainingCount, 1, (INTEGER)_NextCount)
THEN
PROC_AMP_CalculateMultiplicative(_Crystal, _Altar, _CrystalType, _Player, _NewChance, _DiminishingRate, _NextCount);

// success: give matrix, bump counter
PROC
PROC_AMP_RollForSuccess((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_Chance)
AND
Random(100, (INTEGER)_Roll)
AND
_Roll <= _Chance
AND
DB_AMP_Altar_Output((ITEMROOT)_OutputTemplate)
THEN
RequestDelete(_Crystal);
TemplateAddTo(_OutputTemplate, _Player, 1, 1);
ApplyStatus(_Altar, "ARCANIZER_ALTAR_SUCCESS_VFX", 0.0, 1);
PROC_AMP_IncrementSuccessCount(_Altar, _CrystalType);
// PROC_AMP_CheckFirstTimeSuccess(_Player);

// fail: delete crystal
PROC
PROC_AMP_RollForSuccess((ITEM)_Crystal, (ITEM)_Altar, (STRING)_CrystalType, (CHARACTER)_Player, (INTEGER)_Chance)
AND
Random(100, (INTEGER)_Roll)
AND
_Roll > _Chance
THEN
RequestDelete(_Crystal);

// bump success counter
PROC
PROC_AMP_IncrementSuccessCount((ITEM)_Altar, (STRING)_CrystalType)
AND
DB_AMP_Altar_Successes(_Altar, _CrystalType, (INTEGER)_CurrentCount)
AND
IntegerSum(_CurrentCount, 1, (INTEGER)_NewCount)
THEN
NOT DB_AMP_Altar_Successes(_Altar, _CrystalType, _CurrentCount);
DB_AMP_Altar_Successes(_Altar, _CrystalType, _NewCount);

// first success for this type
PROC
PROC_AMP_IncrementSuccessCount((ITEM)_Altar, (STRING)_CrystalType)
AND
NOT DB_AMP_Altar_Successes(_Altar, _CrystalType, _)
THEN
DB_AMP_Altar_Successes(_Altar, _CrystalType, 1);
