Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// tickhara combat barks, ultra pool boss

IF
EnteredCombat((CHARACTER)_Mob, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_06_171ecf1c-df60-427b-a38b-09607737482d)
AND
NOT DB_AMP_Tickhara_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Tickhara_Greeting_528af5fa-af2f-4efd-a755-dc877793317e, _Mob)
THEN
DB_AMP_Tickhara_Greeted(_Mob);
DB_AMP_Tickhara_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Tickhara_BarkUnlock", 5000);

IF
TurnStarted((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_06_171ecf1c-df60-427b-a38b-09607737482d)
AND
NOT DB_AMP_Tickhara_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Tickhara_Greeting_528af5fa-af2f-4efd-a755-dc877793317e, _Mob)
THEN
DB_AMP_Tickhara_Greeted(_Mob);
DB_AMP_Tickhara_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Tickhara_BarkUnlock", 5000);

IF
UsingSpell((CHARACTER)_Mob, _, _, _, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_06_171ecf1c-df60-427b-a38b-09607737482d)
AND
IsDead(_Mob, 0)
AND
NOT DB_AMP_Tickhara_BarkLock(_Mob)
AND
Random(3, (INTEGER)_Roll)
AND
_Roll == 0
THEN
DB_AMP_Tickhara_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Tickhara_BarkDelay", 1500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Tickhara_BarkDelay")
AND
DB_AMP_Tickhara_BarkLock(_Mob)
AND
IsDead(_Mob, 0)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Tickhara_SpellCast_ae395e5d-af9a-4f30-92d5-a70be932b684, _Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Tickhara_BarkUnlock", 10000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Tickhara_BarkUnlock")
AND
DB_AMP_Tickhara_BarkLock(_Mob)
THEN
NOT DB_AMP_Tickhara_BarkLock(_Mob);

IF
HitpointsChanged((CHARACTER)_Mob, (REAL)_Pct)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_06_171ecf1c-df60-427b-a38b-09607737482d)
AND
_Pct <= 35.0
AND
NOT DB_AMP_Tickhara_LowHP(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Tickhara_LowHP_570fca76-02c2-4aef-9fb7-e7f57b3bef3d, _Mob)
THEN
DB_AMP_Tickhara_LowHP(_Mob);

IF
Died((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_06_171ecf1c-df60-427b-a38b-09607737482d)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Tickhara_Death_bcd02476-fefd-4835-8bc2-d6ca7854e2fe, _Mob)
THEN
DB_NOOP(1);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Tickhara_Greeted(_Mob)
THEN
NOT DB_AMP_Tickhara_Greeted(_Mob);
DB_AMP_Tickhara_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Tickhara_BarkUnlock", 5000);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Tickhara_BarkLock(_Mob)
THEN
NOT DB_AMP_Tickhara_BarkLock(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Tickhara_LowHP(_Mob)
THEN
NOT DB_AMP_Tickhara_LowHP(_Mob);

EXITSECTION

ENDEXITSECTION
