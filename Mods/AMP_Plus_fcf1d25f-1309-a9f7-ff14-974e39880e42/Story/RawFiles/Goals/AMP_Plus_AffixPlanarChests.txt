PROC
PROC_AMP_Plus_Chests_Apply()
AND
QRY_AMP_Plus_Affix_On("PLANARCHESTS")
AND
DB_AMP_GameMode((STRING)_Mode)
AND
DB_AMP_Plus_ChestCfg(_Mode, (STRING)_Level, (INTEGER)_Min, (INTEGER)_Max)
THEN
PROC_AMP_Lootbox_SetConfig(_Level, _Min, _Max);

// off stops new spawns only. STOCK is never zeroed, its economy keeps the base counts

PROC
PROC_AMP_Plus_Chests_Apply()
AND
NOT QRY_AMP_Plus_Affix_On("PLANARCHESTS")
AND
NOT DB_AMP_GameMode("STOCK")
AND
DB_AMP_Plus_ChestCfg("NORMAL", (STRING)_Level, _, _)
THEN
PROC_AMP_Lootbox_SetConfig(_Level, 0, 0);

// base re-asserts its table from the same mode write, level start reapplies after it

IF
DB_AMP_GameMode(_)
THEN
PROC_AMP_Plus_Chests_Apply();

IF
LevelGameplayStarted(_, _)
THEN
PROC_AMP_Plus_Chests_Apply();

PROC
PROC_AMP_Plus_Affix_Reconcile((STRING)_Id, (INTEGER)_Val)
AND
_Id == "PLANARCHESTS"
THEN
PROC_AMP_Plus_Chests_Apply();
