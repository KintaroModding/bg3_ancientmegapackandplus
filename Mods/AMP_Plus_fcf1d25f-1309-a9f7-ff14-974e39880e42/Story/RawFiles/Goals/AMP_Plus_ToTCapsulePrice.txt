// swapped row has a shop template with no pairing, so the drain skips it
PROC
PROC_AMP_Plus_ToTCapsulePrice_DrainCapsules()
AND
DB_ToT_ItemFreq((STRING)_ShopID, "CAPSULES", (ITEMROOT)_Base, (INTEGER)_Weight)
AND
DB_AMP_Plus_CapsuleAltOf(_Base, (ITEMROOT)_Alt)
THEN
NOT DB_ToT_ItemFreq(_ShopID, "CAPSULES", _Base, _Weight);
DB_ToT_ItemFreq(_ShopID, "CAPSULES", _Alt, _Weight);
PROC_AMP_Plus_ToTCapsulePrice_DrainCapsules();

PROC
PROC_AMP_Plus_ToTCapsulePrice_DrainCoins()
AND
DB_ToT_ItemFreq((STRING)_ShopID, "COINS", (ITEMROOT)_Base, (INTEGER)_Weight)
AND
DB_AMP_Plus_CoinAltOf(_Base, (ITEMROOT)_Alt)
THEN
NOT DB_ToT_ItemFreq(_ShopID, "COINS", _Base, _Weight);
DB_ToT_ItemFreq(_ShopID, "COINS", _Alt, _Weight);
PROC_AMP_Plus_ToTCapsulePrice_DrainCoins();
