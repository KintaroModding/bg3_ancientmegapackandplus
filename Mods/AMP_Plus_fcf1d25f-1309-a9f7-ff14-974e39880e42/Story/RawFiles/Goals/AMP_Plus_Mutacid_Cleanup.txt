// base leftcombat sweep misses non-party AI allies. life status clears at combat end and deletes the slime
// base cleanup still runs, the owner-link guard makes the second one a no-op

IF
StatusApplied((CHARACTER)_Slime, "AMP_AcidSlime_Control", _, _)
AND
DB_AMP_Mutacid_SlimeOwner(_Slime, _)
THEN
ApplyStatus(_Slime, "AMP_MUTACID_SLIME_LIFE", -1.0, 1, _Slime);

IF
StatusRemoved((CHARACTER)_Slime, "AMP_MUTACID_SLIME_LIFE", _, _)
AND
DB_AMP_Mutacid_SlimeOwner(_Slime, (CHARACTER)_Owner)
THEN
NOT DB_AMP_Mutacid_SlimeOwner(_Slime, _Owner);
RequestDeleteTemporary(_Slime);

// orphan sweep: last-kill acid can seed a slime after the leftcombat sweeps ran
// delay the delete, instant delete of a materializing CreateAt is unreliable
IF
DB_AMP_Mutacid_SlimeOwner((CHARACTER)_Slime, (CHARACTER)_Owner)
AND
IsInCombat(_Owner, 0)
THEN
RealtimeObjectTimerLaunch(_Slime, "AMP_Plus_SlimeOrphan", 1500);

IF
CombatEnded((GUIDSTRING)_Combat)
AND
DB_AMP_Mutacid_SlimeOwner((CHARACTER)_Slime, (CHARACTER)_Owner)
AND
IsInCombat(_Slime, 0)
AND
IsInCombat(_Owner, 0)
THEN
RealtimeObjectTimerLaunch(_Slime, "AMP_Plus_SlimeOrphan", 1500);

IF
LevelGameplayStarted(_, _)
AND
DB_AMP_Mutacid_SlimeOwner((CHARACTER)_Slime, (CHARACTER)_Owner)
AND
IsInCombat(_Owner, 0)
THEN
RealtimeObjectTimerLaunch(_Slime, "AMP_Plus_SlimeOrphan", 1500);

IF
ObjectTimerFinished((CHARACTER)_Slime, "AMP_Plus_SlimeOrphan")
AND
DB_AMP_Mutacid_SlimeOwner(_Slime, (CHARACTER)_Owner)
AND
IsInCombat(_Owner, 0)
THEN
NOT DB_AMP_Mutacid_SlimeOwner(_Slime, _Owner);
RequestDeleteTemporary(_Slime);

// purge by template, no owner link. a retracted link leaves a 0 hp hostile that cannot die and holds combat open
// LeaveCombat and SetOnStage(0) first in case the delete is deferred

IF
EnteredCombat((CHARACTER)_Slime, _)
AND
DB_AMP_Mutacid_AmoebaTemplate((GUIDSTRING)_Template)
AND
GetTemplate(_Slime, _Template)
AND
NOT DB_AMP_Mutacid_SlimeOwner(_Slime, _)
THEN
LeaveCombat(_Slime);
SetOnStage(_Slime, 0);
RequestDeleteTemporary(_Slime);

IF
StatusApplied((CHARACTER)_Slime, "TEMPORARILY_HOSTILE", _, _)
AND
DB_AMP_Mutacid_AmoebaTemplate((GUIDSTRING)_Template)
AND
GetTemplate(_Slime, _Template)
AND
NOT DB_AMP_Mutacid_SlimeOwner(_Slime, _)
THEN
LeaveCombat(_Slime);
SetOnStage(_Slime, 0);
RequestDeleteTemporary(_Slime);

IF
StatusApplied((CHARACTER)_Slime, "TEMPORARILY_HOSTILE", _, _)
AND
DB_AMP_Mutacid_SlimeOwner(_Slime, _)
THEN
RemoveStatus(_Slime, "TEMPORARILY_HOSTILE", NULL_00000000-0000-0000-0000-000000000000);

// panic hatch, SE console: Osi.TextEvent("AMP_MutacidPanic")
// frees players from a frozen combat the event purge cannot reach
IF
TextEvent("AMP_MutacidPanic")
AND
DB_Players((CHARACTER)_P)
THEN
LeaveCombat(_P);
