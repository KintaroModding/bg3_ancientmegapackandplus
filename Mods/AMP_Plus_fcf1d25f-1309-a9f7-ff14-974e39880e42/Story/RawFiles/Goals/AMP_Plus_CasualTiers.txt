// base writes NONE in its Casual branch of PROC_AMP_SetGameMode, rewritten after rather than stopped

PROC
PROC_AMP_Plus_CasualTiers_Assert()
AND
DB_AMP_GameMode("CASUAL")
AND
DB_AMP_Difficulty_Mode("NONE")
THEN
NOT DB_AMP_Difficulty_Mode("NONE");
DB_AMP_Difficulty_Mode("CASUAL");

PROC
PROC_AMP_Plus_CasualTiers_Assert()
AND
DB_AMP_GameMode("STOCK")
AND
DB_AMP_Difficulty_Mode("CASUAL")
THEN
NOT DB_AMP_Difficulty_Mode("CASUAL");
DB_AMP_Difficulty_Mode("NONE");

IF
DB_AMP_GameMode(_)
THEN
PROC_AMP_Plus_CasualTiers_Assert();

IF
LevelGameplayStarted(_, _)
THEN
PROC_AMP_Plus_CasualTiers_Assert();

// ledger records kills under any mode but NONE, so Casual kills are taken back as written or they cash out on a switch to Normal

IF
DB_AMP_BossProgression_Applied((STRING)_Key)
AND
DB_AMP_GameMode("CASUAL")
THEN
NOT DB_AMP_BossProgression_Applied(_Key);

IF
DB_AMP_BossProgression_KillTier((STRING)_Key, (INTEGER)_Tier)
AND
DB_AMP_GameMode("CASUAL")
THEN
NOT DB_AMP_BossProgression_KillTier(_Key, _Tier);
