IF
LevelGameplayStarted(_Level, _)
AND

AND

THEN

AND

AND

THEN

// combat start
IF

AND
DB_KELCIDON_Cat(_Cat)
THEN
RemoveStatus(_Cat, "KELCIDON_PREBATTLE_INVULNERABILITY", NULL_00000000-0000-0000-0000-000000000000);
PlayEffect(_Cat, VFX_Passive_Interrupt_Projected_Ward_01_33c3c5b9-ca4f-0e29-22df-238ee5b7fbc1, "BodyFX", 1.0);
ApplyStatus(_Cat, "KELCIDON_BOSS_STATS", -1.0, 1, _Cat);
ApplyStatus(_Cat, "KELCIDON_DIVINE_WRATH", -1.0, 1, _Cat);
ApplyStatus(_Cat, "KELCIDON_WATCHER_AURA", -1.0, 1, _Cat);
ApplyStatus(_Cat, "KELCIDON_NIGHTMARE_ENGINE", -1.0, 1, _Cat);

PROC_KELCIDON_ApplyNightmareAll(_Cat);

// wall show/hide
PROC
PROC_KELCIDON_ShowWalls()
AND
DB_KELCIDON_Wall((ITEM)_Wall)
THEN
SetOnStage(_Wall, 1);

PROC
PROC_KELCIDON_HideWalls()
AND
DB_KELCIDON_Wall((ITEM)_Wall)
THEN
SetOnStage(_Wall, 0);

// nightmare application (replaces aura)
PROC
PROC_KELCIDON_ApplyNightmareAll((CHARACTER)_Cat)
AND
DB_Players((CHARACTER)_Player)
AND
HasActiveStatus(_Player, "KELCIDON_NIGHTMARE", 0)
AND
HasActiveStatus(_Player, "KELCIDON_AWAKENED", 0)
THEN
ApplyStatus(_Player, "KELCIDON_NIGHTMARE", -1.0, 1, _Cat);

// pulse each cat turn: reapply nightmare + afk check
IF
StatusApplied(_Cat, "KELCIDON_NIGHTMARE_PULSE", _, _)
AND
DB_KELCIDON_Cat((CHARACTER)_Cat)
THEN
PROC_KELCIDON_ApplyNightmareAll(_Cat);
PROC_KELCIDON_DragAFK();

// drag afk players (no damage 2+ turns) to the cat
PROC
PROC_KELCIDON_DragAFK()
AND
DB_Players((CHARACTER)_Player)
AND
IsDead(_Player, 0)
AND
HasActiveStatus(_Player, "KELCIDON_PLAYER_DAMAGED", 0)
THEN
PlayEffect(_Player, VFX_Script_DestroyAction_Arcane_Gate_01_cde28800-1d22-76f0-fa45-06a56f7c367b, "Root", 1.0);
TeleportToPosition(_Player, -740.677, 0.0, 591.152, "", 1, 1, 1, 0, 1);

PROC
PROC_KELCIDON_RemoveNightmareAll()
AND
DB_Players((CHARACTER)_Player)
AND
HasActiveStatus(_Player, "KELCIDON_NIGHTMARE", 1)
THEN
RemoveStatus(_Player, "KELCIDON_NIGHTMARE", NULL_00000000-0000-0000-0000-000000000000);
RemoveStatus(_Player, "KELCIDON_AWAKENED", NULL_00000000-0000-0000-0000-000000000000);

// guard: don't reapply if already humiliated. same stackid without stacktype
// would overwrite -> statusremoved -> death rule -> die.
PROC
PROC_KELCIDON_Humiliate((CHARACTER)_Cat)
AND
NOT DB_KELCIDON_Humiliated(1)
THEN
ApplyStatus(_Cat, "KELCIDON_HUMILIATED", 54.0, 1, NULL_00000000-0000-0000-0000-000000000000);
NOT DB_KELCIDON_Humiliated(0);
DB_KELCIDON_Humiliated(1);

// counter reset on damage
IF
StatusApplied((GUIDSTRING)_Cat, "KELCIDON_DAMAGED_THIS_TURN", _, _)
AND
DB_KELCIDON_Cat((CHARACTER)_Cat)
AND
DB_KELCIDON_Humiliated(1)
AND
HasActiveStatus(_Cat, "KELCIDON_HUMILIATED", 1)
THEN
DB_KELCIDON_Resetting(1);
RemoveStatus(_Cat, "KELCIDON_HUMILIATED", NULL_00000000-0000-0000-0000-000000000000);

// after reset removal -> reapply fresh + flash.
// don't clear resetting here, death rule also fires on statusremoved
IF
StatusRemoved((GUIDSTRING)_Cat, "KELCIDON_HUMILIATED", _, _)
AND
DB_KELCIDON_Cat((CHARACTER)_Cat)
AND
DB_KELCIDON_Resetting(1)
THEN
ApplyStatus(_Cat, "KELCIDON_HUMILIATED", 54.0, 1, NULL_00000000-0000-0000-0000-000000000000);
ApplyStatus(_Cat, "KELCIDON_PRIDE_RESTORED", 6.0, 1, NULL_00000000-0000-0000-0000-000000000000);
PlayEffect(_Cat, VFX_Passive_Interrupt_Projected_Ward_01_33c3c5b9-ca4f-0e29-22df-238ee5b7fbc1, "BodyFX", 1.0);

// clear resetting flag once fresh humiliated is applied
IF
StatusApplied((GUIDSTRING)_Cat, "KELCIDON_HUMILIATED", _, _)
AND
DB_KELCIDON_Cat((CHARACTER)_Cat)
AND
DB_KELCIDON_Resetting(1)
THEN
NOT DB_KELCIDON_Resetting(1);

// death trigger. gated: cat in combat (cleanup removals shouldn't kill) and
// at least one player in combat (no afk-kill from camp).
IF
StatusRemoved((GUIDSTRING)_Cat, "KELCIDON_HUMILIATED", _, _)
AND
DB_KELCIDON_Cat((CHARACTER)_Cat)
AND
DB_KELCIDON_Humiliated(1)
AND
NOT DB_KELCIDON_Resetting(1)
AND
IsInCombat(_Cat, 1)
AND
DB_Players((CHARACTER)_Player)
AND
IsInCombat(_Player, 1)
THEN
Die((CHARACTER)_Cat, DEATHTYPE.Disintegrate, NULL_00000000-0000-0000-0000-000000000000, 1, 0);
