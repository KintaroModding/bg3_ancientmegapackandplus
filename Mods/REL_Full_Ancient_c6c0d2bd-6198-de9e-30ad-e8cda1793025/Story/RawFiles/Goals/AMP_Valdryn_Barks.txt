Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// valdryn combat barks (pool boss)

IF
EnteredCombat((CHARACTER)_Mob, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_04_4c553ed6-ee7a-4598-9d66-1bc064c83f50)
AND
NOT DB_AMP_Valdryn_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Valdryn_Greeting_1f541cd0-acf8-4ec6-a4fd-1c1774ec3c69, _Mob)
THEN
DB_AMP_Valdryn_Greeted(_Mob);
DB_AMP_Valdryn_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Valdryn_BarkUnlock", 5000);

IF
TurnStarted((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_04_4c553ed6-ee7a-4598-9d66-1bc064c83f50)
AND
NOT DB_AMP_Valdryn_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Valdryn_Greeting_1f541cd0-acf8-4ec6-a4fd-1c1774ec3c69, _Mob)
THEN
DB_AMP_Valdryn_Greeted(_Mob);
DB_AMP_Valdryn_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Valdryn_BarkUnlock", 5000);

IF
UsingSpell((CHARACTER)_Mob, _, _, _, _)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_04_4c553ed6-ee7a-4598-9d66-1bc064c83f50)
AND
IsDead(_Mob, 0)
AND
NOT DB_AMP_Valdryn_BarkLock(_Mob)
AND
Random(3, (INTEGER)_Roll)
AND
_Roll == 0
THEN
DB_AMP_Valdryn_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Valdryn_BarkDelay", 1500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Valdryn_BarkDelay")
AND
DB_AMP_Valdryn_BarkLock(_Mob)
AND
IsDead(_Mob, 0)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Valdryn_SpellCast_45ee27b6-f028-46fb-9943-7a64300d3d24, _Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Valdryn_BarkUnlock", 10000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Valdryn_BarkUnlock")
AND
DB_AMP_Valdryn_BarkLock(_Mob)
THEN
NOT DB_AMP_Valdryn_BarkLock(_Mob);

IF
HitpointsChanged((CHARACTER)_Mob, (REAL)_Pct)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_04_4c553ed6-ee7a-4598-9d66-1bc064c83f50)
AND
_Pct <= 35.0
AND
NOT DB_AMP_Valdryn_LowHP(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Valdryn_LowHP_79a9754c-f15e-4dda-b824-cc1cb7f60928, _Mob)
THEN
DB_AMP_Valdryn_LowHP(_Mob);

IF
Died((CHARACTER)_Mob)
AND
GetTemplate(_Mob, AMP_ULTRA_Summon_04_4c553ed6-ee7a-4598-9d66-1bc064c83f50)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Valdryn_Death_dfa2cc8b-7611-4633-a718-bb4cb0f19d07, _Mob)
THEN
DB_NOOP(1);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Valdryn_Greeted(_Mob)
THEN
NOT DB_AMP_Valdryn_Greeted(_Mob);
DB_AMP_Valdryn_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Valdryn_BarkUnlock", 5000);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Valdryn_BarkLock(_Mob)
THEN
NOT DB_AMP_Valdryn_BarkLock(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Valdryn_LowHP(_Mob)
THEN
NOT DB_AMP_Valdryn_LowHP(_Mob);

EXITSECTION

ENDEXITSECTION
