// main rule - hooks the bossrewards questupdate event
IF
QuestUpdateUnlocked((CHARACTER)_Player, "HIDDEN_WLD_Boosters", (STRING)_BossTag)
AND
DB_AMP_GameMode("CASUAL")
AND
DB_AMP_BossCoins(_BossTag, (ITEMROOT)_CoinTemplate, (INTEGER)_Count)
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsCharacter(_Host, 1)
THEN
TemplateAddTo(_CoinTemplate, _Host, _Count, 0);

// grym has no questupdate in bossrewards, hook his death directly
IF
Died(S_UND_KethericCity_AdamantineGolem_2a5997fc-5f2a-4a13-b309-bed16da3b255)
AND
DB_AMP_GameMode("CASUAL")
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsCharacter(_Host, 1)
AND
QRY_OnlyOnce("AMP_BossCoins_Grym")
THEN
TemplateAddTo((ITEMROOT)AMP_FeyCoin_c304469f-6132-4304-a5a2-882a2820be3b, _Host, 1, 0);

EXITSECTION
