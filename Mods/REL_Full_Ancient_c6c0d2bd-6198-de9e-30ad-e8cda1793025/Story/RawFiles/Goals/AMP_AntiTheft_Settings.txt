Version 1
SubGoalCombiner SGC_AND
INITSECTION
DB_AMP_AntiTheft_Disabled(0);

KBSECTION
// anti-theft active check

// hard: always on, toggle ignored
QRY
QRY_AMP_AntiTheft_Active()
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")
THEN
DB_NOOP(1);

// normal: on unless disabled
QRY
QRY_AMP_AntiTheft_Active()
AND
DB_AMP_Difficulty_Mode("STANDARD")
AND
NOT DB_AMP_AntiTheft_Disabled(1)
THEN
DB_NOOP(1);

// settings ring toggle

// spell trigger - non-stock
IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Settings_AntiTheft", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
NOT DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_AntiTheft_ShowToggle(_Caster);

// spell trigger - stock, reuse generic stock message
IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Settings_AntiTheft", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
DB_AMP_GameMode("STOCK")
THEN
OpenMessageBox(_Caster, "AMP_Settings028");

// normal mode, currently on
PROC
PROC_AMP_AntiTheft_ShowToggle((CHARACTER)_Host)
AND
DB_AMP_Difficulty_Mode("STANDARD")
AND
NOT DB_AMP_AntiTheft_Disabled(1)
THEN
DB_AMP_AntiTheft_Pending(_Host);
OpenMessageBoxYesNo(_Host, "AMP_AntiTheft_AskDisable");

// normal mode, currently off
PROC
PROC_AMP_AntiTheft_ShowToggle((CHARACTER)_Host)
AND
DB_AMP_Difficulty_Mode("STANDARD")
AND
DB_AMP_AntiTheft_Disabled(1)
THEN
DB_AMP_AntiTheft_Pending(_Host);
OpenMessageBoxYesNo(_Host, "AMP_AntiTheft_AskEnable");

// hard mode - locked info
PROC
PROC_AMP_AntiTheft_ShowToggle((CHARACTER)_Host)
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")
THEN
OpenMessageBox(_Host, "AMP_AntiTheft_HardLocked");

// casual / no mode - inactive
PROC
PROC_AMP_AntiTheft_ShowToggle((CHARACTER)_Host)
AND
DB_AMP_Difficulty_Mode("NONE")
THEN
OpenMessageBox(_Host, "AMP_AntiTheft_Inactive");

// yes = keep current state (safe default)
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AntiTheft_AskDisable", 1)
AND
DB_AMP_AntiTheft_Pending(_Host)
THEN
NOT DB_AMP_AntiTheft_Pending(_Host);
PROC_AMP_PlayModeEffect(_Host);

IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AntiTheft_AskEnable", 1)
AND
DB_AMP_AntiTheft_Pending(_Host)
THEN
NOT DB_AMP_AntiTheft_Pending(_Host);
PROC_AMP_PlayModeEffect(_Host);

// no = flip state

// on -> off
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AntiTheft_AskDisable", 0)
AND
DB_AMP_AntiTheft_Pending(_Host)
AND
DB_AMP_AntiTheft_Disabled(0)
THEN
NOT DB_AMP_AntiTheft_Pending(_Host);
NOT DB_AMP_AntiTheft_Disabled(0);
DB_AMP_AntiTheft_Disabled(1);
PROC_AMP_PlayModeEffect(_Host);

// off -> on
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_AntiTheft_AskEnable", 0)
AND
DB_AMP_AntiTheft_Pending(_Host)
AND
DB_AMP_AntiTheft_Disabled(1)
THEN
NOT DB_AMP_AntiTheft_Pending(_Host);
NOT DB_AMP_AntiTheft_Disabled(1);
DB_AMP_AntiTheft_Disabled(0);
PROC_AMP_PlayModeEffect(_Host);

// clear stale pending on save load
IF
SavegameLoaded()
AND
DB_AMP_AntiTheft_Pending(_Host)
THEN
NOT DB_AMP_AntiTheft_Pending(_Host);

EXITSECTION
ENDEXITSECTION
