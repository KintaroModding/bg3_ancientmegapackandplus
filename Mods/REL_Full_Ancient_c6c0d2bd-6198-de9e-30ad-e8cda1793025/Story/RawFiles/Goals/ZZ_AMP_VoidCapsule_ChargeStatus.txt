KBSECTION

// re-apply status whenever the charge count db changes
IF
DB_AMP_VoidCapsule_Charges((ITEM)_Capsule, (INTEGER)_Count)
AND
_Count > 0
AND
DB_AMP_VoidCapsule_ChargeToDuration(_Count, (REAL)_Duration)
THEN
ApplyStatus(_Capsule, "AMP_VOID_CAPSULE_CHARGES", _Duration, 1, _Capsule);

// strip status on depletion
IF
DB_AMP_VoidCapsule_Depleted((ITEM)_Capsule)
THEN
RemoveStatus(_Capsule, "AMP_VOID_CAPSULE_CHARGES", _Capsule);
