// level band -> coin template
DB_AMP_PostCombatCoin_Range(1, 4, (ITEMROOT)AMP_FeyCoin_c304469f-6132-4304-a5a2-882a2820be3b);
DB_AMP_PostCombatCoin_Range(5, 9, (ITEMROOT)AMP_ShadowfellCoin_67131136-f11d-4b27-9017-23162629310d);
DB_AMP_PostCombatCoin_Range(10, 99, (ITEMROOT)AMP_AstralCoin_6696e8d1-d87f-401d-a4b3-ce2966ea5b80);

KBSECTION

// casual: 5% planar coin after combat, silent, type by host level

// host leaves combat -> roll
IF
LeftCombat((CHARACTER)_Host, _)
AND
GetHostCharacter(_Host)
AND
IsCharacter(_Host, 1)
AND
DB_AMP_GameMode("CASUAL")
AND
Random(100, (INTEGER)_Roll)
AND
_Roll < 5
THEN
PROC_AMP_PostCombatCoin_Grant(_Host);

// grant coin by level range
PROC
PROC_AMP_PostCombatCoin_Grant((CHARACTER)_Host)
AND
GetLevel(_Host, (INTEGER)_Level)
AND
DB_AMP_PostCombatCoin_Range((INTEGER)_Min, (INTEGER)_Max, (ITEMROOT)_Coin)
AND
_Level >= _Min
AND
_Level <= _Max
THEN
TemplateAddTo(_Coin, _Host, 1, 0);

EXITSECTION
ENDEXITSECTION
